def clean_env(**kw):

    return env

def seed(tag):
    N[0] += 1
    leaves = [{"k": f"m:{SIDE}:{tag}:{N[0]}", "d": {"n": N[0]}, "h": "0" * 64}]
    as_of = datetime(2026, 10, 25, tzinfo=timezone.utc) + timedelta(seconds=N[0])
    root = hashlib.sha256(f"m{SIDE}{tag}{N[0]}".encode()).hexdigest()
    sha = hashlib.sha256(an.manifest(int((as_of - an.EPOCH).total_seconds() * 1e6), leaves, root)).hexdigest()
    bid = adm.execute("insert into anchor_batches (as_of, leaves, merkle_root, manifest_sha256, created_by) values"
                      " (%s, %s::jsonb, %s, %s, 'agent:anchor') returning id",
                      (as_of, json.dumps(leaves), root, sha)).fetchone()[0]
    resp = b"\xf0" + an._put_varuint(16) + b"\x22" * 16 + b"\x08" + b"\x00" + an.ATT_PENDING + \
        an._put_varuint(len(an._put_varuint(len(CAL)) + CAL.encode())) + an._put_varuint(len(CAL)) + CAL.encode()
    adm.execute("insert into anchor_receipts (batch_id, provider, status, calendar, receipt, created_by) values"
                " (%s, 'opentimestamps', 'da_gui', %s, %s, 'agent:anchor')", (bid, CAL, an.ots_file(bytes.fromhex(sha), resp)))
    return bid

    return json.loads(line[-1][10:]) if line else {"harness_err": r.stderr.decode("utf-8", "replace")[-300:]}

def tables():
    return [r[0] for r in adm.execute("select table_name from information_schema.tables where table_schema = 'public'"
                                      " and table_type = 'BASE TABLE' order by 1")]


def cols(t):
    return [r[0] for r in adm.execute("select column_name from information_schema.columns where table_schema ="
                                      " 'public' and table_name = %s order by ordinal_position", (t,))]


def thash(t, cl):
    q = ", ".join(f'"{c}"' for c in cl)
    return adm.execute(f'select count(*), md5(coalesce(string_agg(x, \'|\' order by x), \'\')) from'
                       f' (select row({q})::text as x from public."{t}") s').fetchone()


def snap(colmap=None):
    tabs = tables()
    colmap = colmap or {t: cols(t) for t in tabs}
    return {
        "tables": {t: list(thash(t, colmap[t])) for t in colmap if t in tabs},
        "colmap": colmap,
        "receipts": [list(r) for r in adm.execute(
            "select id, batch_id, status, encode(sha256(coalesce(receipt, ''::bytea)), 'hex'), created_by"
            " from anchor_receipts order by id")],
        "chain_tail": list(adm.execute("select id, row_hash from agent_action_log order by id desc limit 1").fetchone()
                           or []),
        "verify": adm.execute("select count(*) from verify_action_log()").fetchone()[0],
        "view_khoan": list(adm.execute("select count(*), md5(coalesce(string_agg(row(channel, order_id_hash, txn_type,"
                                       " order_amount_vnd, payout_vnd, settled_on, n_ban)::text, '|' order by"
                                       " order_id_hash), '')) from v_settlement_khoan").fetchone()),
        "grants": sorted(list(g) for g in adm.execute(
            "select grantee, table_name, privilege_type from information_schema.role_table_grants where"
            " table_schema = 'public' and grantee <> 'postgres'")),
        "secdef": sorted(list(r) for r in adm.execute(
            "select p.proname, coalesce(array_to_string(p.proconfig, ','), ''), has_function_privilege('public',"
            " p.oid, 'EXECUTE') from pg_proc p join pg_namespace n on n.oid = p.pronamespace where n.nspname ="
            " 'public' and p.prosecdef")),
    }
