class Srv:
    def __init__(self, route):
        self.route, self.req = route, []
        me = self

        class H(BaseHTTPRequestHandler):
            def _any(self):
                u = urllib.parse.urlsplit(self.path)
                n = int(self.headers.get("Content-Length") or 0)
                body = self.rfile.read(n).decode("utf-8") if n else ""
                q = {k: v[0] for k, v in urllib.parse.parse_qs(u.query, keep_blank_values=True).items()}
                me.req.append({"m": self.command, "p": u.path, "q": q, "body": body,
                               "h": {k.lower(): v for k, v in self.headers.items()}})
                st, payload, hd = me.route(self.command, u.path, q, body)
                data = payload if isinstance(payload, bytes) else json.dumps(payload).encode()
                self.send_response(st)
                for k, v in (hd or {}).items():
                    self.send_header(k, v)
                self.send_header("Content-Length", str(len(data)))
                self.end_headers()
                self.wfile.write(data)
            do_GET = do_POST = do_PUT = do_DELETE = _any

            def log_message(self, *a):
                pass
        self.h = ThreadingHTTPServer(("127.0.0.1", 0), H)
        self.url = f"http://127.0.0.1:{self.h.server_address[1]}"
        threading.Thread(target=self.h.serve_forever, daemon=True).start()

    def close(self):
        self.h.shutdown()
        self.h.server_close()

    def paths(self, p):
        return [r for r in self.req if r["p"] == p]

def shopee_route(orders, per_page=2, fail=None, more_forever=False):
    """orders: [(sn, payout, release_ts, income_dict)]; fail(method, path, q, n_seen) → (status, payload, hdr) | None."""
    seen = {}

    def route(m, p, q, body):
        key = (p, q.get("page_no"), q.get("order_sn"))
        seen[key] = seen.get(key, 0) + 1
        if fail:
            r = fail(m, p, q, seen[key])
            if r:
                return r
        if p == ESC_LIST:
            pn = int(q.get("page_no") or 1)
            chunk = orders[(pn - 1) * per_page: pn * per_page]
            more = more_forever or pn * per_page < len(orders)
            return 200, {"error": "", "message": "", "request_id": "rq",
                         "response": {"more": more, "escrow_list": [
                             {"order_sn": sn, "payout_amount": pay, "escrow_release_time": rel}
                             for sn, pay, rel, _inc in chunk]}}, None
        if p == ESC_DET:
            for sn, _pay, _rel, inc in orders:
                if sn == q.get("order_sn"):
                    return 200, {"error": "", "message": "", "request_id": "rq",
                                 "response": {"order_sn": sn, "buyer_user_name": "nguoimuagia",
                                              "return_order_sn_list": [], "order_income": inc}}, None
            return 200, {"error": "order_not_found", "message": "x"}, None
        if p == "/api/v2/auth/access_token/get":
            return 200, {"error": "", "message": "", "access_token": "A_NEW_SECRET_R15",
                         "refresh_token": "R_NEW_SECRET_R15", "expire_in": 14400}, None
        return 404, {"error": "error_not_found"}, None
    return route

@contextlib.contextmanager
def seams(map_file=None, live=False):
    saved = {}
    if sa is not None:
        if map_file is not None:
            saved["MAP_FILE"] = sa.MAP_FILE
            sa.MAP_FILE = Path(map_file)
        if live:
            saved["source_kind"] = sa.source_kind
            sa.source_kind = lambda ch, base: "api_live"
    try:
        yield
    finally:
        for k, v in saved.items():
            setattr(sa, k, v)


def cli(argv, mod=ma):
    o, e = io.StringIO(), io.StringIO()
    with contextlib.redirect_stdout(o), contextlib.redirect_stderr(e):
        try:
            rc = mod.main(argv)
        except SystemExit as ex:
            rc = ex.code if isinstance(ex.code, int) else 2
    text = o.getvalue() + e.getvalue()
    ALL_OUT.append(text)
    return rc, text

def tt_route(stmts, txs, currency="VND"):
    """stmts: [[ (sid, ts) ...] theo trang]; txs: {sid: [[tx...] theo trang]}."""
    def route(m, p, q, body):
        if p == TT_ST:
            i = int(q.get("page_token") or 0)
            page = stmts[i]
            return 200, {"code": 0, "message": "Success", "request_id": "rq", "data": {
                "next_page_token": str(i + 1) if i + 1 < len(stmts) else "",
                "statements": [{"id": sid, "statement_time": ts, "currency": currency, "settlement_amount": "0",
                                "payment_status": "PAID"} for sid, ts in page]}}, None
        parts = p.split("/")
        if len(parts) == 6 and parts[1] == "finance" and parts[4] and parts[5] == "statement_transactions":
            sid = parts[4]
            pages = txs.get(sid, [[]])
            i = int(q.get("page_token") or 0)
            return 200, {"code": 0, "message": "Success", "request_id": "rq", "data": {
                "next_page_token": str(i + 1) if i + 1 < len(pages) else "", "id": sid, "currency": currency,
                "status": "SETTLED", "create_time": 1759449600, "transactions": pages[i]}}, None
        return 404, {"code": 1, "message": "x"}, None
    return route


def ttx(tid, typ="ORDER", oid=None, amt=100000):
    t = {"id": tid, "type": typ, "settlement_amount": str(amt - 15000), "revenue_amount": str(amt),
         "revenue_breakdown": {"subtotal_before_discount_amount": str(amt)},
         "fee_tax_amount": "-5000", "fee_tax_breakdown": {"fee": {"platform_commission_amount": "-4000",
                                                                  "transaction_fee_amount": "-1000"}},
         "shipping_cost_amount": "-10000", "adjustment_amount": "0"}
    if typ == "ORDER":
        t["order_id"] = oid or f"TT{tid}FAKE"
    else:
        t["adjustment_order_id"] = oid or ""
    return t
