"""HARNESS r12 — quét secret: mọi GIÁ TRỊ bí mật thật của lượt (mật khẩu/DSN/muối/khóa vault trong scratchpad r12/secret/*.sh
+ ui*/secrets.json) + mẫu chung (DSN có mật khẩu, `password=` khác ***, `sk-or`) trong evidence/itacc_r12/** và mọi tệp repo
đang đổi/mới (git status). In SỐ lần khớp + tệp, KHÔNG in giá trị.  python -I secret_scan_r12.py.txt"""
import json
import re
import subprocess
import sys
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8")
REPO = Path(__file__).resolve().parents[3]
SP = Path(r"C:\Users\hahuy\AppData\Local\Temp\claude\D--code-samsam\86c80dd1-dfb9-4a98-ba2d-8954bd0353d0\scratchpad\r12")
vals = set()
for f in (SP / "secret").glob("*.sh"):
    for m in re.finditer(r"='([^']+)'", f.read_text(encoding="utf-8")):
        v = m.group(1)
        vals.add(v)
        pm = re.search(r"password=([^ ]+)", v) or re.search(r"://[^:/@]+:([^@]+)@", v)
        if pm:
            vals.add(pm.group(1))
for f in SP.glob("ui*/secrets.json"):
    d = json.loads(f.read_text(encoding="utf-8"))
    for k in ("ui_pw", "db_pw"):
        vals.update(d.get(k, {}).values())
    for k in ("svc_dsn", "owner_dsn"):
        pm = re.search(r"password=([^ ]+)", d.get(k, ""))
        if pm:
            vals.add(pm.group(1))
vals = {v for v in vals if len(v) >= 12}
PAT = re.compile(r"postgres(?:ql)?://[^:/@\s]+:(?!\*\*\*@)[^@\s]+@|password=(?!\*\*\*)[A-Za-z0-9_\-]{12,}|sk-or-[A-Za-z0-9]")
st = subprocess.run(["git", "status", "--porcelain", "--untracked-files=all"], cwd=REPO, capture_output=True,
                    encoding="utf-8", errors="replace").stdout.splitlines()
files = {REPO / ln[3:].strip().strip('"') for ln in st if ln[3:].strip()}
files |= set((REPO / "evidence" / "itacc_r12").rglob("*"))
skip = (".agents", "founder", "lv2-market-top.png")
hits = {}
for f in sorted(files):
    rel = f.relative_to(REPO).as_posix()
    if not f.is_file() or rel.startswith(skip) or f.stat().st_size > 50_000_000:
        continue
    try:
        t = f.read_bytes().decode("utf-8", "replace")
    except OSError:
        continue
    nv = sum(t.count(v) for v in vals)
    lines = [i + 1 for i, ln in enumerate(t.splitlines()) if PAT.search(ln)]
    if nv or lines:
        hits[rel] = (nv, lines)
real = {k: v for k, v in hits.items() if v[0]}
print(f"giá trị bí mật kiểm: {len(vals)} · tệp quét: {len(files)} · tệp khớp GIÁ TRỊ THẬT: {len(real)}")
for k, (nv, lines) in hits.items():
    print(("GIA_TRI_THAT" if nv else "MAU_CHUNG (soát tay: placeholder/ví dụ)") + f" {k} giá_trị={nv} dòng_mẫu={lines[:8]}")
sys.exit(1 if real else 0)
