def files_text(d):
    d = Path(d)
    return "\n".join(f.read_text(encoding="utf-8") for f in sorted(d.glob("*.md"))) if d.exists() else ""


def section(text, head):
    i = text.find(head)
    if i < 0:
        return ""
    j = text.find("\n## ", i + 3)
    return text[i:j if j > 0 else len(text)]
