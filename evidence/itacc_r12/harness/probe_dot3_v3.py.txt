def files(d, pre=""):
    d = Path(d)

def blocked_ok(rc, so, se, fid):

# ================================================================ L2 + RB kd_quote
INTAKE = r"""
import json, sys
from datetime import date
from connectors import b2b_form
from ingest import crm
from ingest.lead_store import connect
from urllib.parse import urlencode
sku, qty, org, phone = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4]
f = {"org_name": org, "org_type": "doanh_nghiep", "region": "Đà Nẵng", "contact_name": "Người Liên Hệ Thử R12",
     "phone": phone, "email": "", "need_by": "", "consent": "b2b", "policy_version": "b2b-v0-nhap", "website": "",
     "sku1": sku, "qty1": qty}
form, code = b2b_form.parse_form(urlencode(f).encode(), [sku], date(2026, 10, 23))
assert form is not None, code
conn = connect()
rid, new = b2b_form.intake(conn, form, crm.load_vault_key(), today=date(2026, 10, 23))
print(json.dumps({"rid": rid, "new": new}))
"""


def intake(tag, sku, qty, org, phone):
    rc, so, se, _ = p.run(f"s_b2b_{tag}", [PY, "-c", INTAKE, sku, qty, org, phone])
    assert rc == 0, se[-400:]
    return json.loads(so.strip().splitlines()[-1])["rid"]


def approve_prices(sku):
    for pid, in p.rows("select id from price_list where sku_key = %s and channel = 'shopee' and approved_by is null"
                       " and valid_to is null order by id", (sku,)):
        assert p.run(f"s_duyet_gia_{pid}", [PY, "-m", "ingest.golden_store", "--approve-price", pid, "--by",
                                            "Người Duyệt Giá Thử R12"])[0] == 0


try:
    r_cho = intake("120", "mat-ong-sam-120ml-1hu", "50", "Công ty Thử R12 Giá Chưa Duyệt", "0900001201")
    rc, so, se, _ = p.run("l2_quote_gia_chua_duyet", [PY, "-m", "agents.kd_quote", "--quote", r_cho, "--dry-run",
                                                      "--date", HOM_NAY])
    p.check("L2", "quote_khong_dung_gia_chua_duyet", "685.000" not in so and "745.000" not in so
            and "thieu_gia_niem_yet" in so and "Traceback" not in se,
            "giá list shopee chưa duyệt → không có đơn giá suy từ JSON (745.000 → 685.000), lý do thieu_gia_niem_yet",
            {"exit": rc, "stdout": so.strip()[-400:]})
    approve_prices("mat-ong-sam-300ml")
    r_ok = intake("300", "mat-ong-sam-300ml", "50", "Công ty Thử R12 Giá Đã Duyệt", "0900001202")
    rc, so, se, _ = p.run("l2_quote_gia_da_duyet", [PY, "-m", "agents.kd_quote", "--quote", r_ok, "--dry-run",
                                                    "--date", HOM_NAY])
    m = re.search(r"CK (\d+)% · đơn giá ([\d.]+)đ", so)
    rnd = json.loads((Path(ROOT) / "data" / "b2b_pricing.json").read_text(encoding="utf-8"))["rounding_vnd"]
    exp = None
    if m:
        ck = int(m.group(1))
        exp = 3100000 * (100 - ck) // (100 * rnd) * rnd
    got = int(m.group(2).replace(".", "")) if m else None
    p.check("L2", "quote_gia_db_da_duyet (positive control)", got is not None and got == exp,
            "đơn giá = giá list price_list ĐÃ duyệt (3.100.000, khác JSON 2.980.000) × (1 − CK)",
            {"exit": rc, "don_gia": got, "ky_vong": exp, "stdout": so.strip()[-300:]})
    rc, so, se, _ = p.run("l2_catalog", [PY, "-m", "agents.kd_catalog", "--build", "--date", HOM_NAY,
                                         "--out-dir", EXT / "cat"])
    md = (EXT / "cat" / f"catalog-tet-{HOM_NAY}.md")
    txt = md.read_text(encoding="utf-8") if md.exists() else ""
    p.check("L2", "catalog_loai_CAM_gia_db (positive control)", rc == 0 and "3.100.000" in txt,
            "catalog có SKU TP_THUONG với giá DB đã duyệt", {"exit": rc, "co_3100": "3.100.000" in txt})
    p.check("L2", "catalog_khong_co_sku_CAM_va_gia_chua_duyet", "745.000" not in txt and "1.340.000" not in txt
            and "2.980.000" not in txt,
            "không SKU readiness CAM (120ml), không giá JSON/chưa duyệt", {"exit": rc, "files": files(EXT / "cat"),
                                                                          "stdout": so.strip()[-300:]})
    q0 = p.q("select count(*) from approval_queue where block = 'sales' and object_type in ('b2b_quote', 'b2b_quote_matrix')")
    tat("sales-05")
    rc, so, se, _ = p.run("rb_kd_quote_tat", [PY, "-m", "agents.kd_quote", "--quote", r_ok, "--submit", "--date",
                                              HOM_NAY, "--out-dir", EXT / "q_tat"])
    q1 = p.q("select count(*) from approval_queue where block = 'sales' and object_type in ('b2b_quote', 'b2b_quote_matrix')")
    b = blocked_ok(rc, so, se, "sales-05")
    p.check("RB", "kd_quote_tat_khong_file_khong_muc", all(b.values()) and not files(EXT / "q_tat") and q1 == q0,
            "exit 2 + [rollout] sales-05, 0 file, 0 mục", {**b, "exit": rc, "files": files(EXT / "q_tat"), "muc": [q0, q1],
                                                         "stderr_cuoi": se.strip()[-200:]})
except Exception as e:  # noqa: BLE001
    p.error("L2", "setup", e)

# ================================================================ F031-6
try:
    assert p.run("s_ar_load", [PY, "-m", "agents.acc_ar", "--load", F / "ar_ok_r10.csv"])[0] == 0
    env = {"PROBE_SET": f"agents.report:REPORTS_DIR={EXT / 'ar'}", "PROBE_CALL_MAIN": "1"}
    rc, so, se, _ = p.run("f0316_bulletin", [PY, "-m", "agents.acc_ar", "--bulletin", "--as-of", "2026-10-23"], env=env, wrap=True)
    fs = files(EXT / "ar", "director-finance-")
    txt = (EXT / "ar" / fs[0]).read_text(encoding="utf-8") if fs else ""
    sec = txt.split("## Công nợ đại lý quá hạn", 1)[-1].split("\n## ", 1)[0] if txt else ""
    head = next((ln for ln in sec.splitlines() if ln.startswith("| Đại lý")), "")
    cols = [c.strip() for c in head.strip("|").split("|")]
    num = lambda s: int(s.strip().strip("`").replace(".", "").replace("đ", "") or 0)  # noqa: E731
    bad, nrow = [], 0
    for ln in sec.splitlines():
        if ln.startswith("| `"):
            cells = [c.strip() for c in ln.strip("|").split("|")]
            row = dict(zip(cols, cells))
            ages = [num(row[k]) for k in cols if k in ("0-30", "31-60", "61-90", ">90")]
            nrow += 1
            if sum(ages) != num(row["Quá hạn"]):
                bad.append(ln)
    p.check("F031-6", "co_cot_0_30_va_cong_khop", rc == 0 and "0-30" in cols and nrow > 0 and not bad,
            "có cột 0-30; mỗi dòng tổng cột tuổi = Quá hạn", {"exit": rc, "cot": cols, "so_dong": nrow, "dong_lech": bad[:3]})
except Exception as e:  # noqa: BLE001
    p.error("F031-6", "setup", e)

OPS_HEAD = "## Chi so ngân sách vận hành"


def ops_section(txt):
