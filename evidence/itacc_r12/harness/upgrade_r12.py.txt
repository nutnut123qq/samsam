DB riêng samsam_itacc_r12_<mã> trên container r12 (KHÔNG samsam). Không in DSN/mật khẩu.
  python -I upgrade_r12.py.txt <base_tree> <repo> <admin_env> <db>
Kiểm: (1) dữ liệu/đuôi hash chain không đổi; (2) quyền chỉ THÊM đúng tập r12 (svc_ui SELECT cột plots/plot_logs +
SELECT chunks; app_human EXECUTE plot_log_add_human), không BỚT; (3) function mới/đổi: SECURITY DEFINER, search_path cố
định, không PUBLIC EXECUTE; link/unlink_lot_origin dùng human_actor; (4) hành vi sau nâng cấp bằng login THẬT (không
owner): svc_ui đọc plots/chunks được, không ghi plot_logs; login người khối farming ghi qua function được, đúng tên;
login người khối sales / ghi tên khác / tên agent bị chặn."""

def run(tree, *args, ok=(0,)):
    p = subprocess.run([sys.executable, *args], cwd=tree, env=ENV, capture_output=True, encoding="utf-8",
                       errors="replace", timeout=900, check=False)
    tail = [x for x in (p.stdout + p.stderr).splitlines() if x.strip()][-2:]
    print(f"[run:{'nen' if tree == BASE else 'repo'}] {' '.join(args)} -> exit {p.returncode}")
    for x in tail:
        print("    " + x[:200])
    if p.returncode not in ok:
        raise SystemExit(1)


def q(sql, a=None, dsn=DSN):
    with psycopg.connect(dsn, connect_timeout=5) as c:
        cur = c.execute(sql, a)
        return cur.fetchall() if cur.description else []


# --- 1. DB nền (mã baseline) + lịch sử
run(BASE, "scripts/apply_schema.py", "--roles")

q("update feature_rollout set stage = 'tat', stage_since = now() - interval '30 days' where feature_id = 'sales-07.1'")

PRIV = ("select grantee, table_name, privilege_type from information_schema.role_table_grants where table_schema ="
        " 'public' and grantee <> current_user union select grantee, table_name || '(' || column_name || ')',"
        " privilege_type from information_schema.column_privileges where table_schema = 'public' and grantee <>"
        " current_user union select grantee, 'function ' || routine_name, privilege_type from"
        " information_schema.routine_privileges where routine_schema = 'public' and grantee <> current_user and"
        " grantee <> 'PUBLIC'")


def counts():
    tabs = [r[0] for r in q("select tablename from pg_tables where schemaname='public' order by 1")]
    return {t: q(f'select count(*) from public."{t}"')[0][0] for t in tabs}

# --- 3. hành vi sau nâng cấp — login THẬT (không owner)

def as_(k, sql, a=None):
    dd = conninfo.conninfo_to_dict(DSN)
    dd.update(user=names[k], password=pw[k])
    return q(sql, a, dsn=conninfo.make_conninfo(**dd))


def denied(k, sql, a=None):
    try:
        as_(k, sql, a)
    except psycopg.Error as e:
        return e.sqlstate
    return None


ADD = "select plot_log_add_human(%s, now() - interval '1 hour', 'tưới nước', 'nâng cấp r12', %s)"
try:
    n_pl = as_("svc", "select count(*) from plot_logs")[0][0]
    n_ck = as_("svc", "select count(*) from chunks")[0][0]

    assert st == "42501"
    st = denied("svc", ADD, (plot, "Vườn Nâng Cấp R12"))
    print(f"  svc_ui gọi plot_log_add_human → {st}")
    assert st == "42501"
    lid = as_("vuon", ADD, (plot, "Vườn Nâng Cấp R12"))[0][0]
    au = q("select author, row_hash is not null from plot_logs where id = %s", (lid,))[0]
    print(f"  login người khối farming ghi qua function → #{lid} author={au[0]!r} hash={au[1]}")
    assert au == ("Vườn Nâng Cấp R12", True)
    for k, by, why in (("sales", "Bán Hàng Nâng Cấp R12", "khối sales"), ("vuon", "Người Khác R12", "ghi tên khác"),

        assert st is not None
