def files(d, pre=""):
    d = Path(d)
    return sorted(f.name for f in d.glob(f"{pre}*")) if d.exists() else []


def blocked_ok(rc, so, se, fid):
    return {"exit_2": rc == 2, "rollout_msg": "[rollout]" in so and fid in so, "khong_traceback": "Traceback" not in se}


def chk_blocked(bug, case, rc, so, se, fid, extra_ok=True, **extra):
    b = blocked_ok(rc, so, se, fid)
    p.check(bug, case, all(b.values()) and extra_ok,
            f"exit 2 + '[rollout] … {fid}', không traceback, không đầu ra/side effect của {fid}",

import json, os, sys
