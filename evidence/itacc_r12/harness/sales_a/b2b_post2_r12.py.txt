"""HARNESS r12 nhóm sales_a — bản chuyển của r10 g2 `b2b_post2.py` (S05-X1): 1 yêu cầu B2B SKU mat-ong-sam-120ml-1hu
(giá list price_list CHƯA duyệt). Đổi so với r10: chờ server bind tối đa 20 giây (POST thử lại khi chưa kết nối được).
Dùng: python b2b_post2_r12.py.txt <url>"""
import sys
import time
from urllib.parse import urlencode

import httpx

sys.stdout.reconfigure(encoding="utf-8")
f = {"org_name": "Công ty Thử IT G2 Probe", "org_type": "doanh_nghiep", "region": "Đà Nẵng",
     "contact_name": "Người Liên Hệ Thử IT", "phone": "0900000302", "email": "", "need_by": "",
     "consent": "b2b", "policy_version": "b2b-v0-nhap", "website": "", "sku1": "mat-ong-sam-120ml-1hu", "qty1": "50"}
t_end = time.time() + 20
while True:
    try:
        r = httpx.post(sys.argv[1], content=urlencode(f).encode(),
                       headers={"Content-Type": "application/x-www-form-urlencoded"}, timeout=10)
        break
    except httpx.ConnectError:
        if time.time() > t_end:
            raise
        time.sleep(0.5)
print(r.status_code, "Đã nhận yêu cầu" in r.text)
