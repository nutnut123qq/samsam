"""r12 (bản dựng từ setup_ui_r10_goc.py.txt — giữ nguyên người/dữ liệu mồi r10, THÊM người vùng trồng cho UI-01;
đổi: DB/login/đường dẫn theo đối số). Dựng môi trường UI thật (DB thử samsam_itacc_r10ui + login thật tạm + cấu hình
tạm). Mật khẩu sinh ngẫu nhiên → secrets.json (scratchpad, KHÔNG vào evidence).
Chạy: python -I setup_ui_r12_full.py.txt <root> <owner_env_db_ui> <runtime_dir>"""
import json
import secrets
import shutil
import sys
from pathlib import Path

import psycopg
from psycopg import conninfo

ROOT, RT = Path(sys.argv[1]), Path(sys.argv[3])
sys.path.insert(0, str(ROOT))
from pipelines.auth import hash_password  # noqa: E402

HERE = RT
HERE.mkdir(parents=True, exist_ok=True)
ADMIN = Path(sys.argv[2]).read_text().split("'")[1]
assert conninfo.conninfo_to_dict(ADMIN)["dbname"].startswith("samsam_itacc_r12_ui")
PEOPLE = {   # mã tài khoản giả → (người, khối, vai trò UI)
    "mkt1": ("Marketing UI Một", "marketing", "gd_marketing"),
    "mkt2": ("Marketing UI Hai", "marketing", "gd_marketing"),
    "sales": ("Bán Hàng UI", "sales", "gd_ban_hang"),
    "mismatch": ("Người Lệch UI", None, "gd_marketing"),   # db_login trỏ login mkt2
    "tgd1": ("TGĐ UI", None, "tgd"),   # xem đủ 7 tab; không login DB (không quyết)
    "vuon": ("Vườn UI R12", "farming", "ha_tang"),          # r12 UI-01
    "lech": ("TGĐ Lệch Vườn UI R12", None, "tgd"),        # r12: db_login = login sales (người khác)
}
sec = {"ui_pw": {}, "db_pw": {}, "login": {}}
with psycopg.connect(ADMIN, autocommit=True, connect_timeout=5) as c:
    for k in ("svc", "mkt1", "mkt2", "sales", "vuon"):
        n = f"r12uf_{k}"
        pw = secrets.token_urlsafe(16)
        c.execute(f"drop role if exists {n}")
        grp = "svc_ui" if k == "svc" else "app_human"
        c.execute(f"create role {n} login password '{pw}' nosuperuser in role {grp}")
        sec["db_pw"][k], sec["login"][k] = pw, n
    c.execute("delete from human_logins")
    for k in ("mkt1", "mkt2", "sales", "vuon"):
        person, block, _ = PEOPLE[k]
        c.execute("insert into human_logins (db_login, person, blocks, granted_by)"
                  " values (%s, %s, %s, 'ui-test-r12')", (sec["login"][k], person, [block]))
    # dữ liệu mồi (owner): 2 draft agent khối marketing + 1 mục hàng duyệt khối sales
    for i in range(2):
        c.execute("insert into content_drafts (brief, channel, text, guardrail,"
                  " source, created_by, content_type, block) values (%s, 'facebook',"
                  " %s, '{\"ok\": true}', 'cli', 'agent:mkt02', 'post', 'marketing')",
                  (f"nháp agent UI {i}", f"Nội dung agent thử UI số {i}. Nội dung có sử dụng AI."))
    c.execute("select enqueue_item('price_review', 'ui-q1', 'sales', 'Mục giá thử UI',"
              " 'payload thử', 'low', '{\"ok\": true}'::jsonb, 'none', 'agent:kd_price', 24)")
d = conninfo.conninfo_to_dict(ADMIN)
d.update(user=sec["login"]["svc"], password=sec["db_pw"]["svc"])
sec["svc_dsn"], sec["owner_dsn"] = conninfo.make_conninfo(**d), ADMIN
cfg = HERE / "cfg"
shutil.rmtree(cfg, ignore_errors=True)
shutil.copytree(ROOT / "data" / "directors", cfg / "directors")
for blk, names in (("marketing", ["Marketing UI Một", "Marketing UI Hai", "Người Lệch UI"]),
                   ("sales", ["Bán Hàng UI"])):
    p = cfg / "directors" / f"{blk}.json"
    j = json.loads(p.read_text(encoding="utf-8"))
    j["approvers"] = names
    p.write_text(json.dumps(j, ensure_ascii=False), encoding="utf-8")
users = {}
for k, (person, _blk, role) in PEOPLE.items():
    pw = secrets.token_urlsafe(14)
    sec["ui_pw"][k] = pw
    users[person.casefold()] = {"role": role, "password_hash": hash_password(pw),
                                "db_login": sec["login"].get({"mismatch": "mkt2", "lech": "sales"}.get(k, k))}
(cfg / "users.json").write_text(json.dumps(users, ensure_ascii=False), encoding="utf-8")
(HERE / "secrets.json").write_text(json.dumps(sec, ensure_ascii=False), encoding="utf-8")
print("setup ok:", {k: v[0] for k, v in PEOPLE.items()})
