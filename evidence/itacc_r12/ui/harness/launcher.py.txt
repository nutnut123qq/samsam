class _Fake:
    def __init__(self):
        self.chat = SimpleNamespace(completions=SimpleNamespace(create=self._chat))
        self.embeddings = SimpleNamespace(create=self._emb)

    def _chat(self, **kw):
        return SimpleNamespace(
            usage=SimpleNamespace(prompt_tokens=10, completion_tokens=10),
            choices=[SimpleNamespace(message=SimpleNamespace(
                content="Sâm Sâm gửi lời chào mùa thu tới khách hàng thân thiết."))])

    def _emb(self, **kw):
        n = len(kw["input"]) if isinstance(kw["input"], list) else 1
        return SimpleNamespace(usage=SimpleNamespace(prompt_tokens=1, completion_tokens=0),
                               data=[SimpleNamespace(embedding=[1.0, 0.0])] * n)
