#!/bin/bash
# r20 FIX1 secret scan (bản vòng 1: ../secret_scan.sh.txt; ghi fix1/secret_scan.txt, quét CẢ evidence/cycle_r20_gate gồm fix1/): evidence/cycle_r20_gate + diff từ Base ad0e620 (dòng thêm) + mật khẩu container trong tệp theo dõi/chưa theo dõi.
#   bash secret_scan.sh.txt <SP>
SP="$1"
E=/d/code/samsam/evidence/cycle_r20_gate; OUTF=$E/fix1/secret_scan.txt
cd /d/code/samsam || exit 2
PW=$(sed -nE "s/^export CT_PW='([^']+)'.*/\1/p" "$SP/secret/admin.sh")
[ ${#PW} -ge 20 ] || { echo "không đọc được mật khẩu container"; exit 2; }
{ echo "## secret scan $E + diff từ Base ad0e620 (dòng thêm) — $(date -Iseconds)"
  echo "### grep -rEn 'postgres(ql)?://[^ ]*:[^*@]+@' → (rỗng = đạt)"
  grep -rEn 'postgres(ql)?://[^ ]*:[^*@]+@' "$E"; git diff ad0e620 -- . | grep '^+' | grep -En 'postgres(ql)?://[^ ]*:[^*@]+@'
  echo "### grep -rn 'sk-[o]r' (tiền tố khóa OpenRouter) → (rỗng = đạt)"
  grep -rn 'sk-[o]r' "$E"; git diff ad0e620 -- . | grep '^+' | grep -n 'sk-[o]r'
  echo "### mật khẩu container samsam-ci-repro (sinh ở scratchpad) trong tệp theo dõi + chưa theo dõi (trừ .agents/, founder/) → (rỗng = đạt)"
  { git ls-files; git ls-files --others --exclude-standard | grep -vE '^(\.agents/|founder/)'; } | tr '\n' '\0' | xargs -0 grep -lF -- "$PW" 2>/dev/null
  echo "### đối chứng dương (phải có ≥1): tệp scratchpad chứa mật khẩu"
  grep -lF -- "$PW" "$SP/secret/admin.sh" | wc -l
  echo "EXIT 0"; } > "$SP/secret_scan.tmp"
mv "$SP/secret_scan.tmp" "$OUTF"
cat "$OUTF"
