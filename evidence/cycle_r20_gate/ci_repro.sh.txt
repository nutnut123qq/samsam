#!/bin/bash
# r20 (/cycle 20 lượt 3) — tái hiện job `test` của .github/workflows/ci.yml trên máy dev Windows.
#   bash ci_repro.sh.txt <SP> <base|work|final> <db> <real|shards|probe> [ca_mỗi_lượt]
# Cây SẠCH như actions/checkout: base = `git archive ad0e620`; work = `git archive HEAD` + `git diff HEAD --binary` của repo
# (nội dung LF như checkout CI; không .env, không tệp ignore, không thư mục lạ) — repo git RIÊNG trong scratchpad (có .git,
# HEAD sạch như checkout). DB MỚI <db> trên container samsam-cycle-r20 (postgres:16-alpine; CI dùng service postgres:16),
# DSN/mật khẩu chỉ ở scratchpad. Môi trường như job `test`: chỉ DATABASE_URL (bỏ PG*/OPENROUTER_API_KEY/AUTH_MODE/...).
# Bước = `run:` của job `test` đọc TỪ ci.yml CỦA CÂY ĐÓ, chạy tuần tự, DỪNG ở bước lỗi đầu tiên (như CI). Khác CI (ghi rõ):
# bỏ `pip install -r requirements.txt` (dùng môi trường local: ruff 0.16.10, Python 3.12.7); bước `python -m pytest -q`:
# real = đúng lệnh; shards = chia lượt cùng thứ tự file của --collect-only (Gotcha 0xC0000142 — CI chạy 1 tiến trình).
SP="$1"; WHICH="$2"; DB="$3"; MODE="$4"; TARGET="${5:-600}"
REPO=/d/code/samsam
E=$REPO/evidence/cycle_r20_gate
case "$WHICH:$DB" in base:samsam_ci_base|work:samsam_ci|final:samsam_ci_final) ;; *) echo "cặp cây/DB không hợp lệ"; exit 2;; esac
case "$MODE" in real|shards|probe) ;; *) echo "mode real|shards|probe"; exit 2;; esac
T="$SP/ci_$WHICH"; OUT="$E/ci_repro_$WHICH.txt"
red() { sed -E -e 's#(postgres(ql)?://[^:/@[:space:]]+:)[^@[:space:]]+@#\1***@#g' -e 's/(password=)[^[:space:]'"'"'"]+/\1***/g'; }
ram() { powershell -NoProfile -Command "[int]((Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory/1024)" | tr -d '\r'; }
fp() { { git -C "$REPO" diff HEAD --binary -- . ':(exclude)DONE.md' ':(exclude)CYCLE_LOOP.md'
         git -C "$REPO" ls-files --others --exclude-standard | grep -vE '^(evidence/|\.agents/|founder/|lv2-market-top\.png)' | sort |
           while read -r f; do echo "== $f"; cat "$REPO/$f"; done; } | sha256sum | cut -c1-64; }
for k in $(env | grep -oE '^(PG[A-Z_]*|[A-Z_]*DATABASE_URL)=' | tr -d '='); do unset $k; done
unset AUTH_MODE APPROVAL_ENFORCE DEPLOY LLM_KILL_FEATURES RED_ALERT_NOTIFY CONV_SAMPLE OPENROUTER_API_KEY
export PYTHONIOENCODING=utf-8
[ -e "$T" ] && { echo "cây $T đã có — không ghi đè"; exit 2; }
mkdir -p "$T"
G() { git -C "$T" -c user.name=ci-repro -c user.email=ci-repro@localhost -c core.autocrlf=false -c core.longpaths=true "$@"; }
{
echo "# tái hiện job test ci.yml — $(date -Iseconds) · cây $WHICH · DB $DB · mode $MODE"
echo "# repo HEAD $(git -C "$REPO" rev-parse HEAD) · fp mã đầu = $(fp)"
if [ "$WHICH" = base ]; then git -C "$REPO" -c core.autocrlf=false archive ad0e620 | tar -x -C "$T"; else git -C "$REPO" -c core.autocrlf=false archive HEAD | tar -x -C "$T"; fi
G init -q && G config core.autocrlf false && G add -A -f && G commit -qm "checkout" || exit 3
if [ "$WHICH" != base ]; then
  # diff CHUẨN HÓA theo cấu hình repo (autocrlf) = đúng nội dung commit sẽ lưu (bản đầu dùng autocrlf=false → 3 tệp w/crlf vào cây dạng CRLF)
  git -C "$REPO" diff HEAD --binary > "$SP/work.patch"
  G apply "$SP/work.patch" && G add -A && G commit -qm "work" || { echo "áp diff lỗi"; exit 3; }
fi
want=$(git -C "$REPO" ls-tree -r --name-only HEAD | wc -l)
if [ "$WHICH" = base ]; then src=ad0e620; ck=HEAD; else src=HEAD; ck=HEAD~1; fi
echo "# cây sạch: tree commit checkout $(G rev-parse "$ck^{tree}") vs repo $src $(git -C "$REPO" rev-parse "$src^{tree}") (trùng = byte + mode y hệt checkout CI)"
[ "$WHICH" != base ] && echo "# diff áp vào cây: sha256 $(G diff HEAD~1 HEAD --binary | sha256sum | cut -c1-16) vs work.patch $(sha256sum < "$SP/work.patch" | cut -c1-16)"
echo "# cây sạch $T: $(G ls-files | wc -l) tệp (repo HEAD có $want; tệp thật trên đĩa $(cd "$T" && find . -path ./.git -prune -o -type f -print | wc -l)) · git status --porcelain $(G status --porcelain | wc -l) dòng · .env $( [ -e "$T/.env" ] && echo CÓ || echo không)"
source "$SP/secret/admin.sh"
python -I - "$DB" "$SP/secret/env_$WHICH.sh" <<'PY' || exit 3
import os, sys, time
import psycopg
sys.stdout.reconfigure(encoding="utf-8")
from psycopg import conninfo
db, out = sys.argv[1], sys.argv[2]
d = conninfo.conninfo_to_dict(os.environ["CYCLE_ADMIN"])
assert d["host"] == "127.0.0.1" and d["dbname"] == "postgres"
for _ in range(30):
    try:
        c = psycopg.connect(conninfo.make_conninfo(**d), autocommit=True, connect_timeout=5)
        break
    except psycopg.OperationalError:
        time.sleep(1)
else:
    sys.exit("không kết nối được container")
with c:
    if c.execute("select 1 from pg_database where datname = %s", (db,)).fetchone():
        sys.exit(f"DB {db} đã tồn tại — không drop/ghi đè")
    roles = [r[0] for r in c.execute("select rolname from pg_roles where rolname !~ '^pg_' order by 1")]
    c.execute(f"create database {db}")
print(f"# DB mới {db} @ 127.0.0.1:{d['port']} · role cụm lúc tạo: {roles}")
# DATABASE_URL dạng URL như CI (scheme postgresql, user, mật khẩu, host, port, db); ghép chuỗi để quét bí mật không bắt nhầm mẫu mã
url = "postgresql" + "://" + d["user"] + ":" + d["password"] + "@127.0.0.1:" + str(d["port"]) + "/" + db
open(out, "w").write("export DATABASE_URL='" + url + "'\n")
PY
unset CYCLE_ADMIN CT_PW
source "$SP/secret/env_$WHICH.sh"
python -I - "$T/.github/workflows/ci.yml" > "$SP/steps_$WHICH.txt" 2> "$SP/timeout_$WHICH.txt" <<'PY'
import sys, yaml
# Windows: print mặc định xuống dòng CRLF → bash đọc lệnh dính CR (newline=LF)
sys.stdout.reconfigure(encoding="utf-8", newline=chr(10))
job = yaml.safe_load(open(sys.argv[1], encoding="utf-8"))["jobs"]["test"]
print(f"timeout-minutes {job.get('timeout-minutes')}", file=sys.stderr)
for s in job["steps"]:
    if "run" in s:
        print(s["run"].strip())
PY
echo "# ci.yml của cây: job test $(cat "$SP/timeout_$WHICH.txt") · $(wc -l < "$SP/steps_$WHICH.txt") bước run"
cd "$T" || exit 3
ALL=0; n=0
while IFS= read -r cmd <&3; do
  n=$((n + 1)); t0=$(date +%s)
  case "$cmd" in *$'\r'*) echo "lệnh có CR — dừng"; exit 3;; esac
  case "$cmd" in
    "pip install"*) echo "## bước $n: $cmd → BỎ (môi trường local: $(ruff --version), $(python --version 2>&1))"; continue;;
    "python -m pytest -q")
      if [ "$MODE" = real ]; then
        python -m pytest -q > "$SP/step.log" 2>&1; rc=$?
      elif [ "$MODE" = probe ]; then
        # probe: preflight phiên (sessionstart) + drift-guard ci.yml, không chạy full
        python -m pytest -q tests/test_ci_workflow.py > "$SP/step.log" 2>&1; rc=$?
      else
        python -m pytest --collect-only -q > "$SP/collect_$WHICH.txt" 2>&1; rc=$?
        if [ $rc -eq 0 ]; then
          grep '::' "$SP/collect_$WHICH.txt" | sed 's/::.*//' | uniq -c | awk '{print $2, $1}' > "$E/per_file.txt"
          awk -v t="$TARGET" '{ if (k >= t) { s++; k = 0 } print s+1, $1; k += $2 }' "$E/per_file.txt" > "$E/shards.txt"
          S=$(tail -1 "$E/shards.txt" | cut -d' ' -f1)
          echo "# collect-only: $(tail -1 "$SP/collect_$WHICH.txt") · $(awk '{s+=$2} END {print s}' "$E/per_file.txt") ca / $(wc -l < "$E/per_file.txt") tệp → $S lượt"
          for i in $(seq 1 "$S"); do
            files=$(awk -v i="$i" '$1 == i {print $2}' "$E/shards.txt"); nf=$(echo "$files" | wc -l)
            r0=$(ram); s0=$(date +%s)
            python -m pytest -q -ra $files 2>&1 | red > "$E/shard_$(printf %02d "$i").txt"; sr=${PIPESTATUS[0]}
            s1=$(date +%s); r1=$(ram)
            dll=$(grep -c "3221225794\|0xC0000142" "$E/shard_$(printf %02d "$i").txt")
            echo "lượt $i/$S: $nf tệp · RAM trống $r0→$r1 MB · $((s1 - s0)) s · rc=$sr · 0xC0000142×$dll · $(tail -1 "$E/shard_$(printf %02d "$i").txt" | grep -oE '[0-9]+ (passed|failed|skipped|errors?)[^=]*' | head -1)"
            [ "$sr" -ne 0 ] && rc=1
          done
        fi
        echo "tóm tắt pytest chia lượt" > "$SP/step.log"
      fi;;
    *) bash -c "$cmd" > "$SP/step.log" 2>&1; rc=$?;;
  esac
  echo "## bước $n: $cmd → exit $rc ($(( $(date +%s) - t0 )) s)"; tail -n 6 "$SP/step.log" | red | sed 's/^/    /'
  if [ "$cmd" = "python -m pytest -q" ]; then echo "    # sau pytest: git status --porcelain của cây $(G status --porcelain | wc -l) dòng"; G status --porcelain | head -5 | sed 's/^/    /'; fi
  [ $rc -ne 0 ] && { ALL=$rc; echo "DỪNG ở bước lỗi đầu tiên (như CI — các bước sau skipped)"; break; }
done 3< "$SP/steps_$WHICH.txt"
echo "# fp mã cuối = $(fp)"
echo "EXIT $ALL"
exit "$ALL"
} 2>&1 | red > "$OUT"
exit "${PIPESTATUS[0]}"
