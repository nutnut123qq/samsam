#!/bin/bash
# HARNESS r15 — fingerprint cây bất kỳ (checkout hoặc snapshot baseline) so với baseline 70ff9f7, alternate index (không đụng
# index thật). Mã sản phẩm = diff vs 70ff9f7 + tệp mới trong NEW_FILES_LIST, trừ DONE.md, docs/handover/v13-board.md, evidence/.
#   bash fp_r15.sh.txt <TREE> full|code <NEW_FILES_LIST>      (code = loại thêm **/*.md)
# Tái lập: cùng TREE + cùng danh sách tệp mới → cùng hash; snapshot baseline cho diff rỗng → sha256 của chuỗi rỗng.
TREE="$1"; MODE="$2"; NEW="$3"
SP="<SCRATCHPAD>/r15_fp"
mkdir -p "$SP"
export GIT_INDEX_FILE="$SP/fp.$$.index"
G="git --git-dir=/d/code/samsam/.git --work-tree=$TREE"
rm -f "$GIT_INDEX_FILE"
cd "$TREE" || exit 2
$G read-tree 70ff9f74c3328cf4ecfd7f173d6a0a6a666f6046
$G add -N $(cat "$NEW") 2>/dev/null
EX=(':(exclude)DONE.md' ':(exclude)docs/handover/v13-board.md' ':(exclude)evidence' ':(exclude).gitattributes')
[ "$MODE" = code ] && EX+=(':(exclude,glob)**/*.md')
$G diff 70ff9f74c3328cf4ecfd7f173d6a0a6a666f6046 -- . "${EX[@]}" 2>/dev/null | sha256sum | cut -d' ' -f1
rm -f "$GIT_INDEX_FILE"
