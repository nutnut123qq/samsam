def race(callers, tag):
    """callers: [(nhãn, fn)] chạy song song trên kết nối thật riêng. Barrier chỉ ở ĐẦU; sau SELECT thật của
    cross_source mỗi caller báo đã kiểm rồi chờ caller kia tối đa HOLD giây (timeout → tiếp tục)."""
    start = threading.Barrier(len(callers))
    checked, lock_wait = [], []
    both = threading.Event()

    def cross(conn, ch, hs, kind):
        n = _orig(conn, ch, hs, kind)
        checked.append({"source": kind, "saw_other_source": n, "t": time.monotonic()})
        if len(checked) >= len(callers):
            both.set()
        both.wait(HOLD)
        return n

    def watch():
        t_end = time.monotonic() + HOLD + 6
        while time.monotonic() < t_end and len(lock_wait) == 0:
            if adm.execute("select count(*) from pg_locks where locktype = 'advisory' and not granted"
                           " and database = (select oid from pg_database where datname = current_database())"
                           ).fetchone()[0]:
                lock_wait.append(True)
            time.sleep(0.05)

    def run(fn):
        start.wait(20)
        return fn()

    sc.cross_source = cross
    out = io.StringIO()
    try:
        with contextlib.redirect_stdout(out), concurrent.futures.ThreadPoolExecutor(len(callers) + 1) as pool:
            w = pool.submit(watch)
            futs = [pool.submit(run, fn) for _, fn in callers]
            codes = [f.result(60) for f in futs]
            w.result(30)
    finally:
        sc.cross_source = _orig
    return codes, checked, bool(lock_wait), out.getvalue()
