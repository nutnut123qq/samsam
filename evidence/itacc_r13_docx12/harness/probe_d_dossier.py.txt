"""PROBE r13 D v1 — production-02 kiểm sổ hồ sơ gốc (`ingest.product_dossiers`). Thuần, không DB/mạng. Cùng mã/input/assertion
baseline (1ad90df) và bản sửa.  python -I probe_d_dossier.py.txt <ROOT> <out.json>
Tiêu chí (khóa trước):
- sổ THẬT (repo) → exit 2, báo thiếu nêu ĐÍCH DANH từng SKU golden chưa gắn hồ sơ (13 mã) + "0/14"; repo không bịa hồ sơ.
- hồ sơ thiếu `phien_ban` hoặc `tep_sha256` (sha256 tệp gốc) → lỗi cấu trúc nêu đúng trường.
- hồ sơ đánh dấu `mau: true` → lỗi, KHÔNG được đếm vào số hồ sơ thật.
- hồ sơ đủ trường (gồm phiên bản + hash) + gắn đủ SKU → 0 lỗi, 0 thiếu."""
import contextlib
import io
import json
import sys

ROOT, OUT = sys.argv[1], sys.argv[2]
sys.stdout.reconfigure(encoding="utf-8")
sys.path.insert(0, ROOT)
from ingest import product_dossiers as pdo  # noqa: E402

GOLD = [{"sku_key": "sku-a", "receipt_no": "11/2024/ĐKSP"}, {"sku_key": "sku-b", "receipt_no": None}]


def dossier(**over):
    d = {"ma_ho_so": "HS-01", "ten": "SP A", "thanh_phan": ["sâm"], "quy_cach": "hộp", "so_cong_bo": "11/2024/ĐKSP",
         "loai": "tpbvsk", "sku_keys": ["sku-a"], "xac_nhan_boi": "Người Thử", "can_cu": "BB-thu", "phien_ban": "v1",
         "tep_sha256": "a" * 64}
    d.update(over)
    return d


def doc(items, need=2, rows=None):
    return {"so_ho_so_hop_dong": need, "danh_muc_ho_so_goc": items,
            "sku": rows or [{"sku_key": "sku-a", "ho_so_goc": "HS-01"}, {"sku_key": "sku-b", "ho_so_goc": "HS-02"}]}


rows = []


def case(ma, fn):
    try:
        ok, actual = fn()
    except Exception as e:  # noqa: BLE001
        ok, actual = False, f"{type(e).__name__}: {str(e)[:200]}"
    rows.append({"ca": ma, "ket_qua": "PASS" if ok else "FAIL", "actual": actual})


def c_real():
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf):
        rc = pdo.main(["--check"])
    real = json.loads(pdo.DOSSIER_FILE.read_text(encoding="utf-8"))
    gold = json.loads(pdo.GOLDEN_FILE.read_text(encoding="utf-8"))["skus"]
    t = buf.getvalue()
    return (rc == 2 and real["danh_muc_ho_so_goc"] == [] and "0/14" in t
            and all(g["sku_key"] in t for g in gold)), {"rc": rc, "out": t[-300:]}


def c_field(k):
    def f():
        r = pdo.check(doc([dossier(**{k: ""})], need=1, rows=[{"sku_key": "sku-a", "ho_so_goc": "HS-01"},
                                                               {"sku_key": "sku-b", "ho_so_goc": None}]), GOLD)
        return any(k in e for e in r["errors"]), r["errors"]
    return f


def c_mau():
    r = pdo.check(doc([dossier(mau=True)], need=1), GOLD)
    return r["dossiers"] == 0 and any("MẪU" in e or "mau" in e for e in r["errors"]), r


def c_ok():
    r = pdo.check(doc([dossier(), dossier(ma_ho_so="HS-02", ten="SP B", so_cong_bo="TCB 02", loai="food",
                                          sku_keys=["sku-b"], tep_sha256="b" * 64)]), GOLD)
    return r["errors"] == [] and r["missing"] == [] and r["mapped"] == 2, r


case("so_that_neu_dich_danh_sku", c_real)
case("thieu_phien_ban_loi", c_field("phien_ban"))
case("thieu_tep_sha256_loi", c_field("tep_sha256"))
case("ho_so_mau_khong_tinh", c_mau)
case("du_truong_khong_loi", c_ok)
out = {"probe": "d_dossier_v1", "root": ROOT, "ca": rows,
       "tong": {k: sum(r["ket_qua"] == k for r in rows) for k in ("PASS", "FAIL")}}
open(OUT, "x", encoding="utf-8").write(json.dumps(out, ensure_ascii=False, indent=1, default=str))
for r in rows:
    print(r["ket_qua"], r["ca"], str(r["actual"])[:160])
print("TONG", out["tong"])
sys.exit(1 if out["tong"]["FAIL"] else 0)
