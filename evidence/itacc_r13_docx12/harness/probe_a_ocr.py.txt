"""PROBE r13 A v1 — OCR tại chỗ cho data_platform-02.1 + production-03. Không DB (psycopg.connect bị chặn), CHẶN MẠNG (mọi
socket connect bị từ chối → chứng minh không gửi ảnh ra ngoài/không tự tải model). Cùng mã/input/assertion cho baseline
(1ad90df, registry OCR chỉ `none`) và bản sửa.  python -I probe_a_ocr.py.txt <ROOT> <FIXTURE_DIR> <out.json>
Tiêu chí (khóa trước):
- PNG nhãn có chữ thật → doc_intake.extract_text(..., "paddle_local") ra source=ocr; sx_label.extract trên chữ OCR có ĐKSP
  15/2026/ĐKSP, NSX 2026-10-12, HSD 2028-10-12, LOT L2026-1012, quy cách 300ml; sx_label.read_image dùng đúng provider.
- PDF CHỈ ẢNH 2 trang → source=ocr, pages=2, extract_fields có so_giay 123/2026/ĐKSP; chữ trang 2 có mặt.
- ảnh trắng → source=none (không coi rỗng là thành công); ảnh hỏng → DocError; PDF 31 trang → DocError (trần 30).
- thiếu model cục bộ (OCR_PADDLE_MODEL_DIR trỏ thư mục rỗng) → DocError, 0 kết nối mạng.
- provider ghi engine truy vết (doc_intake.OCR_ENGINES['paddle_local']).
- 0 lần mở socket ra mạng trong cả probe."""
import json
import os
import socket
import sys
import tempfile
from pathlib import Path

ROOT, FX, OUT = sys.argv[1], Path(sys.argv[2]), sys.argv[3]
sys.stdout.reconfigure(encoding="utf-8")
sys.path.insert(0, ROOT)
os.chdir(ROOT)
for k in list(os.environ):
    if k.endswith("DATABASE_URL") or k.startswith("PG") or k in ("OPENROUTER_API_KEY", "OCR_PADDLE_MODEL_DIR"):
        os.environ.pop(k, None)
NET = []
_orig_connect = socket.socket.connect


def _no_net(self, addr):
    NET.append(str(addr)[:60])
    raise OSError("probe: chặn mạng")


socket.socket.connect = _no_net
socket.create_connection = lambda *a, **k: _no_net(None, a[0] if a else None)
import psycopg  # noqa: E402

psycopg.connect = lambda *a, **k: (_ for _ in ()).throw(RuntimeError("probe: cấm DB"))
rows = []


def case(nhom, ma, fn):
    try:
        ok, actual = fn()
    except Exception as e:  # noqa: BLE001
        ok, actual = False, f"{type(e).__name__}: {str(e)[:200]}"
    rows.append({"nhom": nhom, "ca": ma, "ket_qua": "PASS" if ok else "FAIL", "actual": actual})


from ingest import doc_intake as di  # noqa: E402


def ocr(name):
    return di.extract_text(FX / name, "paddle_local", Path(name).suffix)


def c_nhan():
    from agents import sx_label
    text, pages, src = ocr("nhan_mat_ong.png")
    f = sx_label.extract(text)
    got = {"src": src, "dksp": f["dksp"], "nsx": [str(d) for d in f["nsx"]], "hsd": [str(d) for d in f["hsd"]],
           "lots": f["lots"], "qty": f["qty"]}
    ok = (src == "ocr" and any("15/2026" in x for x in f["dksp"]) and "2026-10-12" in got["nsx"]
          and "2028-10-12" in got["hsd"] and "L2026-1012" in f["lots"] and any("300" in q for q in f["qty"]))
    return ok, got


def c_read_image():
    from agents import sx_label
    t = sx_label.read_image(FX / "nhan_mat_ong.png", "paddle_local")
    return "2026" in t, {"len": len(t)}


def c_pdf():
    text, pages, src = ocr("tiep_nhan_scan.pdf")
    f = di.extract_fields(text)
    return (src == "ocr" and pages == 2 and "123/2026" in (f.get("so_giay") or ("",))[0]
            and "r13" in text.lower()), {"src": src, "pages": pages, "fields": {k: v[0] for k, v in f.items()}}


def c_trang():
    text, pages, src = ocr("trang.png")
    return src == "none" and text == "", {"src": src}


def c_err(name):
    def f():
        try:
            ocr(name)
        except di.DocError as e:
            return True, f"DocError: {e}"
        return False, "không lỗi"
    return f


def c_no_model():
    d = tempfile.mkdtemp()
    os.environ["OCR_PADDLE_MODEL_DIR"] = d
    old = getattr(di, "_paddle", None)
    if hasattr(di, "_paddle"):
        di._paddle = None
    n0 = len(NET)
    try:
        ocr("nhan_mat_ong.png")
        return False, "không lỗi"
    except di.DocError as e:
        return len(NET) == n0, f"DocError: {e} · net+{len(NET) - n0}"
    finally:
        os.environ.pop("OCR_PADDLE_MODEL_DIR", None)
        if hasattr(di, "_paddle"):
            di._paddle = old


case("LOI", "nhan_png_ocr_trich_truong", c_nhan)
case("LOI", "sx_label_read_image", c_read_image)
case("LOI", "pdf_scan_2_trang", c_pdf)
case("LOI", "anh_trang_none", c_trang)
case("LOI", "anh_hong_docerror", c_err("hong.jpg"))
case("LOI", "pdf_qua_trang_docerror", c_err("qua_trang.pdf"))
case("LOI", "thieu_model_tu_choi", c_no_model)
case("LOI", "engine_truy_vet", lambda: (bool(getattr(di, "OCR_ENGINES", {}).get("paddle_local")),
                                        getattr(di, "OCR_ENGINES", None)))
case("DOI_CHUNG", "provider_none_giu_cho_ocr", lambda: (di.extract_text(FX / "trang.png", "none", ".png")[2] == "none",
                                                       "none"))
case("DOI_CHUNG", "khong_mo_mang", lambda: (NET == [], NET))
out = {"probe": "a_ocr_v1", "root": ROOT, "ca": rows,
       "tong": {k: sum(r["ket_qua"] == k for r in rows) for k in ("PASS", "FAIL")}}
open(OUT, "x", encoding="utf-8").write(json.dumps(out, ensure_ascii=False, indent=1, default=str))
for r in rows:
    print(r["ket_qua"], r["nhom"], r["ca"], str(r["actual"])[:160])
print("TONG", out["tong"])
sys.exit(1 if out["tong"]["FAIL"] else 0)
