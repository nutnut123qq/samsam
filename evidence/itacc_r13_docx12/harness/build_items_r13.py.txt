def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def rel(p):
    p = Path(p).resolve()
    try:
        return str(p.relative_to(REPO)).replace("\\", "/")
    except ValueError:
        return str(p).replace("\\", "/")
