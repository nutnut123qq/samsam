def env_paddle(intake, model_dir=""):

def intake(env, exit_, *res, out_re=None):
    st = {"argv": ["python", "-m", "ingest.doc_intake"], "env": env, "expect_exit": exit_}
    if res:
        st["stdout_re"] = "(?s)" + "".join(f"(?=.*{r})" for r in res)
    st["out_re"] = out_re or NO_TB
    return st


def prep(dest, *src, name=None, suffix=None):
    a = ["python", PREP, dest, *src]
    if name:
        a += ["--name", name]
    if suffix:
        a += ["--suffix", suffix]
    return {"argv": a, "expect_exit": 0, "stdout_re": "\\[prep\\] ok"}


def allre(*res):
    return "(?s)" + "".join(f"(?=.*{r})" for r in res)
