    st = {"argv": ["python", STT, "--mode", mode, "--model-dir", MODEL],
          "env": {"GARDEN_MEDIA_DIR": MEDIA, "GARDEN_STT_PROVIDER": None, "GARDEN_STT_MODEL_DIR": None,
                  "GARDEN_STT_MAX_SECONDS": None, "GARDEN_STT_TIMEOUT_S": None, **(env_extra or {})},
          "expect_exit": exit_}
    if stdout_re:
        st["stdout_re"] = stdout_re
    if stderr_re:
        st["stderr_re"] = stderr_re
    return st
