def gate(name, pat=None):
    t = (G / name).read_text(encoding="utf-8")
    x = re.findall(r"(?m)^EXIT (\d+)", t)[-1]
    m = re.search(pat, t) if pat else None
    return (m.group(1).strip("= ") if m else ""), x
