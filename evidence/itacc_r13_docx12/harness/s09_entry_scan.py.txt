"""HARNESS r13 sales-09 S09-C01 — bản chuyển TẤT ĐỊNH của lệnh r10 (evidence/itacc_r10/items/sales-09.json S09-C01:
`ls connectors/ api/; grep -rln 'website.*chat|web_chat|webchat|chatbot website' connectors api app; grep CHAT_CHANNELS/
SAN_CHANNELS connectors/channels.py`, kỳ vọng "không có adapter kênh web chat").
Đổi so với r10: grep r10 không có -E nên `|` là ký tự thường (mẫu gần như không thể khớp) → ở đây dùng đúng Ý ĐỊNH của mẫu:
regex Python không phân biệt hoa thường trên TỪNG DÒNG mọi tệp .py dưới connectors/ api/ app/ (bỏ __pycache__) + tên tệp
`web?chat`; CHAT_CHANNELS/SAN_CHANNELS đọc bằng ast (KHÔNG import module — không nạp .env, không DB); danh mục sales-09
`modules` rỗng (tiêu chí r10 T1). Chỉ đọc.  python s09_entry_scan.py.txt
Exit 0 + KHONG_CO_CHATBOT_WEBSITE = không có adapter/entry point chatbot website; 1 = có dấu hiệu."""
import ast
import json
import re
import sys
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8")
ROOT = Path(__file__).resolve().parents[3]
PAT = re.compile(r"website.*chat|web_chat|webchat|chatbot website", re.I)
files, hits, names = [], [], []
for d in ("connectors", "api", "app"):
    for p in sorted((ROOT / d).rglob("*.py")):
        if "__pycache__" in p.parts:
            continue
        rel = p.relative_to(ROOT).as_posix()
        files.append(rel)
        if re.search(r"web.?chat", p.name, re.I):
            names.append(rel)
        for i, ln in enumerate(p.read_bytes().decode("utf-8").split("\n"), 1):
            if PAT.search(ln):
                hits.append(f"{rel}:{i}")
vals = {}
for n in ast.parse((ROOT / "connectors" / "channels.py").read_bytes().decode("utf-8")).body:
    if (isinstance(n, ast.Assign) and len(n.targets) == 1 and isinstance(n.targets[0], ast.Name)
            and n.targets[0].id in ("CHAT_CHANNELS", "SAN_CHANNELS")):
        vals[n.targets[0].id] = ast.literal_eval(n.value)
feats = json.loads((ROOT / "data" / "contract_features.json").read_bytes().decode("utf-8"))["features"]
f09 = next(f for f in feats if f.get("id") == "sales-09")
print("## tệp .py:", files)
print("## dòng khớp mẫu website chat (rỗng = không có):", hits)
print("## tên tệp web chat (rỗng = không có):", names)
print("## kênh:", vals)
print("## danh mục sales-09:", {k: f09.get(k) for k in ("modules", "agent_ids", "code_status")})
ok = (not hits and not names and vals.get("CHAT_CHANNELS") == ("zalo", "messenger")
      and f09.get("modules") == [])
print("KHONG_CO_CHATBOT_WEBSITE" if ok else "CO_DAU_HIEU_CHATBOT_WEBSITE")
sys.exit(0 if ok else 1)
