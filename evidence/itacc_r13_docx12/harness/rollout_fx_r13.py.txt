import sys
from pathlib import Path

REPO = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(REPO))
for s in (sys.stdout, sys.stderr):
    s.reconfigure(encoding="utf-8", errors="replace")
from pipelines import rollout  # noqa: E402

FID, ARGS = sys.argv[1], sys.argv[2:]
_orig = rollout.load_features


def _patched(*a, **k):
    feats = _orig(*a, **k)
    hit = [f for f in feats if f["id"] == FID]
    assert len(hit) == 1, f"không thấy {FID} trong danh mục"
    return [dict(f, code_status="co_code") if f["id"] == FID else f for f in feats]


rollout.load_features = _patched
print(f"[harness] FIXTURE {FID}: code_status → co_code (chỉ trong tiến trình)", file=sys.stderr, flush=True)
sys.exit(rollout.main(ARGS))
