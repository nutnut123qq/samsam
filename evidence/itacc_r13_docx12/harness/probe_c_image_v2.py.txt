REQ, REC = [], []

def png(size, color=(30, 120, 60)):
    b = io.BytesIO()
    Image.new("RGB", size, color).save(b, "PNG")
    return base64.b64encode(b.getvalue()).decode()


def handler(request):
    REQ.append({"url": str(request.url), "body": json.loads(request.content or b"{}"),
                "auth": request.headers.get("authorization", "")[:7]})
    step = STATE["script"].pop(0) if STATE["script"] else ("ok", None)
    kind, arg = step
    if kind == "status":
        return httpx.Response(arg, json={"error": {"code": arg}})
    if kind == "timeout":
        raise httpx.ReadTimeout("probe timeout", request=request)
    if kind == "bad_b64":
        return httpx.Response(200, json={"created": 1, "data": [{"b64_json": "@@@không-phải-base64", "media_type": "image/png"}]})
    w, h = arg or tuple(map(int, REQ[-1]["body"].get("size", "64x64").split("x")))
    return httpx.Response(200, json={"created": 1, "data": [{"b64_json": png((w, h)), "media_type": "image/png"}],
                                     "usage": {"prompt_tokens": 0, "completion_tokens": 4175, "total_tokens": 4175,
                                               "cost": 0.04}})

rows = []


def case(ma, fn, reset=True):
    if reset:
        REQ.clear()
        REC.clear()
        STATE.update(spend=0.0, script=[])
    try:
        ok, actual = fn()
    except Exception as e:  # noqa: BLE001
        ok, actual = False, f"{type(e).__name__}: {str(e)[:200]}"
    rows.append({"ca": ma, "ket_qua": "PASS" if ok else "FAIL", "actual": actual})


def gen(size=(64, 48)):
    return ims.generate_background("nền vườn sâm buổi sáng", size, provider="openrouter_images")


def c_ok():
    img, meta = gen()
    b = REQ[0]["body"] if REQ else {}
    only = gw._config().get("provider_only")
    ok = (img.size == (64, 48) and meta.get("ai_generated") is True and meta.get("model") == "probe/model-gia"
          and meta.get("provider") == "openrouter_images" and len(REQ) == 1 and REQ[0]["url"].endswith("/images")
          and b.get("size") == "64x48" and b.get("model") == "probe/model-gia" and b.get("prompt")
          and b.get("provider", {}).get("only") == only and b["provider"].get("allow_fallbacks") is False
          and set(b) <= {"model", "prompt", "size", "output_format", "n", "provider"}
          and REC == [{"agent": "image", "model": "probe/model-gia", "cost": 0.04, "ok": True,
                       "feature": "marketing-02.1"}])
    return ok, {"meta": meta, "body_keys": sorted(b), "rec": REC, "n_req": len(REQ)}


def c_compose():
    d = Path(tempfile.mkdtemp(prefix="r13img-"))
    ims.make_sample_packshot(d / "pk" / "mat-ong-sam-300ml.png", "mat-ong-sam-300ml")
    res = ims.compose("mat-ong-sam-300ml", "facebook", "Quà biếu mùa thu", fmt="1:1", scene="vườn sâm buổi sáng",
                      readiness={("mat-ong-sam-300ml", "facebook"): "TP_THUONG"}, provider="openrouter_images", packshot_dir=d / "pk", out_dir=d / "out")
    if not res.get("ok"):
        return False, res.get("refused")
    img = Image.open(res["path"]).convert("RGB")
    ok = ims.verify_packshot(img, res["fitted"], res["packshot_xy"]) and res["meta"].get("ai_generated") is True
    return ok, {"ai": res["meta"].get("ai_generated"), "provider": res["meta"].get("provider"), "n_req": len(REQ)}


def blocked(setup, teardown=None):
    def f():
        setup()
        try:
            gen()
            return False, "không chặn"
        except ims.ImageProviderError as e:
            return len(REQ) == 0, f"{e} · req={len(REQ)}"
        finally:
            if teardown:
                teardown()
    return f


def c_retry502():
    STATE["script"] = [("status", 502), ("ok", None)]
    img, meta = gen()
    return len(REQ) == 2 and [r["ok"] for r in REC] == [True], {"n_req": len(REQ), "rec": REC}


def failing(script):
    def f():
        STATE["script"] = list(script)
        try:
            gen()
            return False, "không lỗi"
        except ims.ImageProviderError as e:
            return all(not r["ok"] for r in REC) or script[0][0] == "ok", f"{e} · req={len(REQ)} rec={REC}"
    return f
