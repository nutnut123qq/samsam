def sign(raw, secret=None):
    d = json.loads(raw)
    return "mac=" + hashlib.sha256((d["app_id"] + raw.decode() + str(d["timestamp"])
                                    + (secret or zalo.OA_SECRET)).encode()).hexdigest()


def zpost(uid, text, mid, bad=False):
    ev = {"app_id": zalo.APP_ID, "sender": {"id": uid}, "recipient": {"id": "oa-thu"},
          "event_name": "user_send_text", "message": {"msg_id": mid, "text": text},
          "timestamp": str(int(time.time() * 1000))}
    raw = json.dumps(ev, ensure_ascii=False).encode()
    sig = sign(raw, "sai-secret") if bad else sign(raw)
    return httpx.post(ZURL + "/zalo-webhook", content=raw,
                      headers={"X-ZEvent-Signature": sig}, timeout=10).status_code


def mpost(psid, text, mid, bad=False):
    ev = {"object": "page", "entry": [{"messaging": [{"sender": {"id": psid},
          "message": {"mid": mid, "text": text}}]}]}
    raw = json.dumps(ev, ensure_ascii=False).encode()
    key = b"sai" if bad else os.environ["MESSENGER_APP_SECRET"].encode()
    sig = "sha256=" + hmac.new(key, raw, hashlib.sha256).hexdigest()
    return httpx.post(MURL + "/messenger-webhook", content=raw,
                      headers={"X-Hub-Signature-256": sig,
                               "Content-Type": "application/json"}, timeout=10).status_code


def conv():
    p = zalo.CONV_LOG
    return [json.loads(x) for x in p.read_bytes().split(b"\n") if x.strip()] if p.exists() else []


def calls():
    p = Path(os.environ["IT_FAKE_LOG"])
    return [json.loads(x) for x in p.read_text(encoding="utf-8").splitlines()] if p.exists() else []


def qmax():
    return DB.execute("select coalesce(max(id),0) from approval_queue").fetchone()[0]


def qrows(after):
    return [dict(zip(("id", "block", "object_type", "object_id", "created_by", "status", "title", "payload"), r))
            for r in DB.execute("select id, block, object_type, object_id, created_by, status, title, payload"
                                " from approval_queue where id > %s order by id", (after,)).fetchall()]


def wait_conv(n0, k=1, timeout=20):
    t = time.time() + timeout
    while time.time() < t:
        c = conv()
        if len(c) >= n0 + k:
            time.sleep(0.8)   # để ghi DB sau convlog xong
            return conv()[n0:]
        time.sleep(0.2)
    return conv()[n0:]


def run_case(name, fn):
    n0, c0, q0 = len(conv()), len(calls()), qmax()
    res = {"case": name}
    res.update(fn())
    res["convlog_new"] = conv()[n0:]
    res["fake_calls_new"] = calls()[c0:]
    res["queue_new"] = qrows(q0)
    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / f"sales-01_{name}.json").write_text(json.dumps(res, ensure_ascii=False, indent=1, default=str),
                                               encoding="utf-8")
    print(f"== {name}: http={res.get('http')} conv+{len(res['convlog_new'])} calls+{len(res['fake_calls_new'])}"
          f" queue+{[(q['object_type'], q['created_by']) for q in res['queue_new']]}")
    return res
