def sha(p: Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()


def rel(p: Path) -> str:
    return p.relative_to(R).as_posix()


def lenh(name: str) -> str:
    p = O / f"{name}.txt"
    if not p.exists():
        return name
    m = re.search(r"^# lenh: (.*)$", p.read_text(encoding="utf-8"), re.M)
    return m.group(1).strip() if m else name


def case(ma, mo_ta, files, expected, actual, exit_, kq, ly=None):
    cmds = [lenh(f) for f in files if (O / f"{f}.txt").exists()]
    return {"ma": ma, "mo_ta": mo_ta, "lenh": " ;; ".join(cmds) or "(xem evidence)",
            "expected": expected, "actual": actual, "exit": exit_, "ket_qua": kq,
            "ly_do_not_run": ly, "_files": files}
