HARNESS_COMMON = ["patch_run.py", "force_vanhanh.py"]

crit_exp = {(i, c["ma"]): c for i, it in crit.items() for c in it["ca"]}
summary = []
for fid, cfg in CFG.items():
    k, cr = khung[fid], crit[fid]
    rs = [r for r in results if r["id"] == fid]
    ca, tags = [], []
    for r in rs:
        ce = crit_exp.get((fid, r["ma"]))
        exp = ce["expected"] if ce else "(bổ sung sau khóa — xem actual)"
        ca.append({"ma": r["ma"], "mo_ta": ce["mo_ta"] if ce else "bổ sung sau khóa", "lenh": lenh(r["tag"]) if r["tag"] != "-" else "—",
                   "expected": exp, "actual": r["actual"], "exit": r["exit"], "ket_qua": r["ket_qua"],
                   "ly_do_not_run": r["actual"] if r["ket_qua"] == "NOT_RUN" else ""})
        if r["tag"] != "-":
            tags.append(r["tag"])
    # mọi file out/ bắt đầu bằng tiền tố tag (gồm _a/_b/_off/_on…)
    evs = set()
    for t in tags:
        for f in (EV / "out").glob(t + "*.txt"):
            evs.add(f)
    for e in EXTRA_EV.get(fid, []):
        evs.add(EV / e)
    for r in cfg["rep"]:
        evs.add(EV / "reports" / r)
    for n in cfg["inputs"] + HARNESS_COMMON:
        evs.add(H / n)
    evid = [{"path": rel(p), "sha256": sha(p)} for p in sorted(evs) if p.exists()]
    loi = [L[x] for x in cfg["loi"]]
    nfail = sum(1 for c in ca if c["ket_qua"] == "FAIL")
    assert not (cfg["st"] == "DAT" and nfail), fid
    rec = {"id": fid, "ten": k["ten"], "khoi": k["khoi"], "giai_doan": k["giai_doan"], "pham_vi": k["pham_vi"], "can_cu": k["can_cu"],
           "tieu_chi": cr["tieu_chi_cu_the"], "fingerprint": FP, "thoi_gian": NOW, "moi_truong": cfg["env"],
           "dau_vao": inp(*cfg["inputs"]), "ca": ca, "trang_thai": cfg["st"], "muc_kiem": cfg["mk"],
           "nguyen_nhan": cfg["nn"], "can_de_go": cfg["go"], "gioi_han": cfg["gh"], "evidence": evid, "loi_san_pham": loi,
           "nhom": "g1", "tieu_chi_khoa": "evidence/itacc_r10/criteria/g1.json (sha256 trong g1.lock.txt)"}
    out = REPO / "evidence" / "itacc_r10" / "items" / f"{fid}.json"
    out.write_bytes(json.dumps(rec, ensure_ascii=False, indent=1).encode("utf-8"))
    summary.append((fid, cfg["st"], cfg["mk"], cfg["nn"], len(ca), nfail, len(evid)))
