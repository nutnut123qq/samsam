def db(sql, args=()):

def step(name, ok, page, **kw):
    shot = SHOTS / f"{len(R) + 1:02d}_{name}.png"
    page.screenshot(path=str(shot), full_page=True)

def idle(page, t=1.2):
    page.wait_for_timeout(int(t * 1000))
    for _ in range(60):
        if not page.locator('[data-testid="stStatusWidget"]').is_visible():
            break
        page.wait_for_timeout(500)
    page.wait_for_timeout(400)


def login(page, who, pw=None):
    page.goto(URL)
    page.get_by_label("Tên đăng nhập").wait_for(timeout=60000)
    page.get_by_label("Tên đăng nhập").fill(USER[who])
    page.get_by_label("Mật khẩu", exact=True).fill(pw or SEC["ui_pw"][who])
    page.get_by_role("button", name="Đăng nhập").click()
    idle(page, 2)


def logout(page):
    page.get_by_role("button", name="Đăng xuất").click()
    idle(page)


def tab(page, name):
    page.get_by_role("tab", name=name).click()
    idle(page, 0.8)


def unlock(page, pw):
    page.get_by_label("Mật khẩu Postgres").fill(pw)
    page.get_by_role("button", name="Mở quyền quyết định").click()
    idle(page, 2)


def exp(page, prefix):
    e = page.locator('[data-testid="stExpander"]', has_text=prefix).first
    e.locator("summary").first.click()
    page.wait_for_timeout(500)
    return e


def text(page):
    return page.locator("body").inner_text()


def draft(did):
    """(status, reviewer chuẩn hóa casefold, created_by, created_login, block,
    scheduled_date) — UI ghi reviewer bằng tên đăng nhập (đã casefold)."""
    r = db("select status, reviewer, created_by, created_login, block,"
           " scheduled_date from content_drafts where id = %s", (did,))[0]
    return (r[0], r[1].casefold() if r[1] else r[1], *r[2:])
