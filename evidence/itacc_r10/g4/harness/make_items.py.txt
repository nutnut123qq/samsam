def sha(p: Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()


def rel(p: Path) -> str:

def run_info(ma):
    """(lệnh, exit) từ out/<ma>.txt; ca chỉ truy vấn → lệnh q.py."""
    p = OUT / f"{ma}.txt"
    if not p.exists():
        return None, None
    t = p.read_text(encoding="utf-8")
    m1 = re.search(r"^# lenh (.*)$", t, re.M)
    m2 = re.search(r"^# exit (-?\d+)$", t, re.M)
    if m1:
        return m1.group(1), int(m2.group(1))
    m3 = re.findall(r"^> (.*)$", t, re.M)
    if not m3:
        return f"(script kiểm python inline — kết quả trong out/{ma}.txt)", 0
    return "python evidence/itacc_r10/g4/harness/q.py " + " ; ".join(
        s[:160] for s in m3), 0


def C(ma, mo_ta, expected, actual, kq="PASS", runs=None, note=None):
    runs = runs or [ma]
    cmds, exits = [], []
    for r in runs:
        c, e = run_info(r)
        if c:
            cmds.append(f"[{r}] {c}")
            exits.append(e)
    d = {"ma": ma, "mo_ta": mo_ta, "lenh": " || ".join(cmds) or "(đọc file)",
         "expected": expected, "actual": actual,
         "exit": exits[0] if len(exits) == 1 else exits, "ket_qua": kq,
         "ly_do_not_run": None}
    if note:
        d["ghi_chu"] = note
    return d
