def wcsv(name, header, rows):
    buf = io.StringIO()
    w = csv.writer(buf, lineterminator="\n")
    w.writerow(header)
    w.writerows(rows)
    (FX / name).write_bytes(buf.getvalue().encode("utf-8"))


def wjson(name, obj):
    (FX / name).write_bytes(json.dumps(obj, ensure_ascii=False, indent=1)
                            .encode("utf-8"))
