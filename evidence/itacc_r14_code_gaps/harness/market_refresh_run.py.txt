"""HARNESS r14 (CONTRACT_TESTED, server giả) — data_platform-06: chạy THẬT `python -m ingest.market_api --channel shopee
--kind listing` (tiến trình con, DB thử của runner) với kho token `MARKET_TOKEN_FILE` = <STORE> (NGOÀI repo) đang HẾT HẠN →
CLI phải refresh theo tài liệu chính thức rồi mới kéo listing bằng token mới. Server GIẢ 127.0.0.1: POST
/api/v2/auth/access_token/get (mode ok → token mới; loi → `error_auth`), get_item_list/get_item_base_info (2 item).
Key/token GIẢ (không in giá trị — log server chỉ in sha256[:8] của access_token).
  python market_refresh_run.py.txt ok|loi <STORE>
In: stdout/stderr của CLI, SERVER_LOG, `STORE_ROTATED <bool>` / `STORE_UNCHANGED <bool>`, `LEAK <số chuỗi bí mật lộ ra>`."""
import hashlib
import json
import os
import subprocess
import sys
import threading
import time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from urllib.parse import parse_qs, urlsplit

sys.stdout.reconfigure(encoding="utf-8", errors="replace")
MODE, STORE = sys.argv[1], Path(sys.argv[2])
R_OLD, R_NEW, A_OLD, A_NEW = "R14_REFRESH_OLD_GIA", "R14_REFRESH_NEW_GIA", "R14_ACCESS_OLD_GIA", "R14_ACCESS_NEW_GIA"
PKEY = "r14-partner-key-gia-thu"
SECRETS = (R_OLD, R_NEW, A_OLD, A_NEW, PKEY)
LOG = []
ITEMS = {"27357883504": ("Savina 12v API thử R14", "savina-12v", 175000),
         "24325987926": ("Savigout 30v API thử R14", "savigout-30v", 455000)}


def h8(s):
    return hashlib.sha256(s.encode()).hexdigest()[:8] if s else "-"


class H(BaseHTTPRequestHandler):
    def _send(self, code, body):
        b = json.dumps(body).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(b)))
        self.end_headers()
        self.wfile.write(b)

    def _any(self):
        u = urlsplit(self.path)
        q = {k: v[0] for k, v in parse_qs(u.query).items()}
        n = int(self.headers.get("Content-Length") or 0)
        body = json.loads(self.rfile.read(n) or b"{}") if n else {}
        if u.path == "/api/v2/auth/access_token/get":
            LOG.append(f"{self.command} {u.path} body_refresh={h8(body.get('refresh_token', ''))}")
            if MODE == "ok" and body.get("refresh_token") == R_OLD:
                return self._send(200, {"error": "", "message": "", "access_token": A_NEW, "refresh_token": R_NEW,
                                        "expire_in": 14400, "shop_id": 200000, "partner_id": 100000})
            return self._send(200, {"error": "error_auth", "message": "invalid refresh token (giả)"})
        LOG.append(f"{self.command} {u.path} token={h8(q.get('access_token', ''))}")
        if u.path == "/api/v2/product/get_item_list":
            return self._send(200, {"error": "", "response": {"item": [{"item_id": int(i)} for i in ITEMS],
                                                              "has_next_page": False}})
        if u.path == "/api/v2/product/get_item_base_info":
            ids = q["item_id_list"].split(",")
            return self._send(200, {"error": "", "response": {"item_list": [
                {"item_id": int(i), "item_name": ITEMS[i][0], "item_sku": ITEMS[i][1], "item_status": "NORMAL",
                 "has_model": False, "price_info": [{"original_price": ITEMS[i][2]}],
                 "stock_info_v2": {"summary_info": {"total_available_stock": 7}}} for i in ids]}})
        self.send_response(404)
        self.end_headers()
    do_GET = do_POST = _any

    def log_message(self, *a):
        pass


STORE.parent.mkdir(parents=True, exist_ok=True)
STORE.write_text(json.dumps({"shopee": {"access_token": A_OLD, "refresh_token": R_OLD,
                                        "expire_at": int(time.time()) - 60}}), encoding="utf-8")
before = STORE.read_bytes()
srv = ThreadingHTTPServer(("127.0.0.1", 0), H)
threading.Thread(target=srv.serve_forever, daemon=True).start()
env = {k: v for k, v in os.environ.items() if not k.startswith(("SHOPEE_", "MARKET_"))}
env.update(SHOPEE_API_BASE=f"http://127.0.0.1:{srv.server_address[1]}", SHOPEE_PARTNER_ID="100000",
           SHOPEE_PARTNER_KEY=PKEY, SHOPEE_SHOP_ID="200000", MARKET_TOKEN_FILE=str(STORE), PYTHONIOENCODING="utf-8")
p = subprocess.run([sys.executable, "-m", "ingest.market_api", "--channel", "shopee", "--kind", "listing"],
                   env=env, capture_output=True, text=True, encoding="utf-8", timeout=180,
                   cwd=str(Path(__file__).resolve().parents[3]))
out = p.stdout + p.stderr
print(out)
print("SERVER_LOG:")
print("\n".join(LOG) or "(0 request)")
st = json.loads(STORE.read_text(encoding="utf-8")).get("shopee", {})
print(f"STORE_ROTATED {st.get('access_token') == A_NEW and st.get('refresh_token') == R_NEW}")
print(f"STORE_UNCHANGED {STORE.read_bytes() == before}")
print(f"NEW_TOKEN_SHA8 {h8(A_NEW)}")
print(f"LEAK {sum(s in out for s in SECRETS)}")
srv.shutdown()
raise SystemExit(p.returncode)
