argv = sys.argv[1:]


def pop(flag):
    if flag in argv:
        i = argv.index(flag)
        v = argv[i + 1]
        del argv[i:i + 2]
        return v
    return "none"

def _guard(self, addr):

        NET[0] += 1
        raise OSError("[HARNESS] mạng bị chặn")
    return _orig(self, addr)

    return an._put_varuint(len(b)) + b
