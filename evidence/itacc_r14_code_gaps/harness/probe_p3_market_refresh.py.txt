class Srv:
    def __init__(self, route):
        self.route, self.req = route, []
        srv = self

        class H(BaseHTTPRequestHandler):
            def _any(self):
                u = urllib.parse.urlsplit(self.path)
                n = int(self.headers.get("Content-Length") or 0)
                body = self.rfile.read(n).decode("utf-8") if n else ""
                q = {k: v[0] for k, v in urllib.parse.parse_qs(u.query).items()}
                srv.req.append({"m": self.command, "p": u.path, "q": q, "b": body})
                st, payload, hd = srv.route(self.command, u.path, q, body)
                data = json.dumps(payload).encode()
                self.send_response(st)
                for k, v in (hd or {}).items():
                    self.send_header(k, v)
                self.send_header("Content-Length", str(len(data)))
                self.end_headers()
                self.wfile.write(data)
            do_GET = do_POST = _any

            def log_message(self, *a):
                pass
        self.h = ThreadingHTTPServer(("127.0.0.1", 0), H)
        self.url = f"http://127.0.0.1:{self.h.server_address[1]}"
        threading.Thread(target=self.h.serve_forever, daemon=True).start()

    def close(self):
        self.h.shutdown()
        self.h.server_close()


def sp_route(auth_resp=(200, None, None)):
    def r(m, p, q, b):
        if p == SP_AUTH:
            st, payload, hd = auth_resp
            return st, payload if payload is not None else {
                "error": "", "message": "", "request_id": "x", "access_token": A2, "refresh_token": R2,
                "expire_in": 14400, "shop_id": int(SHOP), "partner_id": int(PID)}, hd
        if p == "/api/v2/product/get_item_list":
            return 200, {"error": "", "message": "", "response": {"item": [], "has_next_page": False}}, None
        return 404, {"error": "not_found"}, None
    return r


def store(path, data):
    path.write_text(json.dumps(data), encoding="utf-8")
    return path.read_bytes()


def run(argv, env, mode="van_hanh"):
    old = dict(os.environ)
    os.environ.update(env)
    rollout._FORCE_MODE = mode
    o, e = io.StringIO(), io.StringIO()
    try:
        with contextlib.redirect_stdout(o), contextlib.redirect_stderr(e):
            try:
                rc = ma.main(argv)
            except SystemExit as x:
                rc = x.code
    finally:
        os.environ.clear()
        os.environ.update(old)
    text = o.getvalue() + e.getvalue()
    return rc, text, [s for s in SECRETS if s in text]


def sp_env(srv, f=None, access=None):
    env = {"SHOPEE_PARTNER_ID": PID, "SHOPEE_PARTNER_KEY": PKEY, "SHOPEE_SHOP_ID": SHOP, "SHOPEE_API_BASE": srv.url}
    if f:
        env["MARKET_TOKEN_FILE"] = str(f)
    if access:
        env["SHOPEE_ACCESS_TOKEN"] = access
    return env

def case(ma_, fn):
    try:
        ok, actual = fn()

        ok, actual = False, f"{type(e).__name__}: {str(e)[:200]}"

def expired(tag):
    f = TMP / f"{tag}.json"
    store(f, {"shopee": {"access_token": AOLD, "refresh_token": R1, "expire_at": int(time.time()) - 60}})
    return f


def c_sp_ok():
    s = Srv(sp_route())
    f = expired("ok")
    try:
        rc, text, leaked = run(["--channel", "shopee", "--refresh"], sp_env(s, f))
    finally:
        s.close()
    st = json.loads(f.read_text(encoding="utf-8")).get("shopee", {})
    rq = [x for x in s.req if x["p"] == SP_AUTH]
    good = False
    if len(rq) == 1:
        q, b = rq[0]["q"], json.loads(rq[0]["b"] or "{}")
        sig = hmac.new(PKEY.encode(), f"{PID}{SP_AUTH}{q.get('timestamp')}".encode(), hashlib.sha256).hexdigest()
        good = (rq[0]["m"] == "POST" and q.get("partner_id") == PID and q.get("sign") == sig
                and "access_token" not in q and b == {"refresh_token": R1, "partner_id": int(PID),
                                                      "shop_id": int(SHOP)})
    ok = (rc == 0 and good and st.get("access_token") == A2 and st.get("refresh_token") == R2
          and abs(st.get("expire_at", 0) - (time.time() + 14400)) < 120 and not leaked and len(s.req) == 1)
    return ok, {"rc": rc, "reqs": [(x["m"], x["p"]) for x in s.req], "leaked": leaked, "text": text[-300:]}


def c_sp_auto():
    s = Srv(sp_route())
    f = expired("auto")
    try:
        rc, text, leaked = run(["--channel", "shopee", "--kind", "listing"], sp_env(s, f))
    finally:
        s.close()
    paths = [x["p"] for x in s.req]
    data = [x for x in s.req if x["p"] != SP_AUTH]
    ok = (rc == 0 and paths and paths[0] == SP_AUTH and data and all(x["q"].get("access_token") == A2 for x in data)
          and not leaked)
    return ok, {"rc": rc, "paths": paths, "leaked": leaked, "text": text[-300:]}


def unchanged(tag, resp, extra=None):
    def f():
        s = Srv(sp_route(resp))
        s2 = Srv(sp_route()) if extra == "redirect" else None
        if s2:
            s.route = lambda m, p, q, b: (302, {}, {"Location": s2.url + SP_AUTH})
        fpath = expired(tag)
        before = fpath.read_bytes()
        try:
            rc, text, leaked = run(["--channel", "shopee", "--kind", "listing"], sp_env(s, fpath))
        finally:
            s.close()
            if s2:
                s2.close()
        n_auth = len([x for x in s.req if x["p"] == SP_AUTH])
        n_data = len([x for x in s.req if x["p"] != SP_AUTH])
        ok = (rc == 1 and fpath.read_bytes() == before and n_auth == 1 and n_data == 0 and not leaked
              and (s2 is None or not s2.req))
        return ok, {"rc": rc, "auth": n_auth, "data": n_data, "redirect_target": len(s2.req) if s2 else None,
                    "leaked": leaked, "text": text[-300:]}
    return f


def c_tt_ok():
    seen = {}

    def r(m, p, q, b):
        seen.setdefault("req", []).append((m, p, q))
        if p == TT_AUTH:
            return 200, {"code": 0, "message": "success", "request_id": "x", "data": {
                "access_token": A2, "access_token_expire_in": int(time.time()) + 604800, "refresh_token": R2,
                "refresh_token_expire_in": int(time.time()) + 86400 * 30}}, None
        return 404, {"code": 1}, None
    s = Srv(r)
    f = TMP / "tt.json"
    store(f, {"tiktok": {"access_token": AOLD, "refresh_token": R1, "expire_at": int(time.time()) - 60}})
    env = {"TIKTOK_APP_KEY": TKEY, "TIKTOK_APP_SECRET": TSEC, "TIKTOK_SHOP_CIPHER": "ROW_c", "TIKTOK_AUTH_BASE": s.url,
           "TIKTOK_API_BASE": s.url, "MARKET_TOKEN_FILE": str(f)}
    try:
        rc, text, leaked = run(["--channel", "tiktok", "--refresh"], env)
    finally:
        s.close()
    st = json.loads(f.read_text(encoding="utf-8")).get("tiktok", {})
    reqs = seen.get("req", [])
    good = (len(reqs) == 1 and reqs[0][0] == "GET" and reqs[0][1] == TT_AUTH and reqs[0][2] == {
        "app_key": TKEY, "app_secret": TSEC, "refresh_token": R1, "grant_type": "refresh_token"})
    ok = rc == 0 and good and st.get("access_token") == A2 and st.get("refresh_token") == R2 and not leaked
    return ok, {"rc": rc, "n": len(reqs), "leaked": leaked, "text": text[-300:]}
