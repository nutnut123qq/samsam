"""HARNESS r14-fix2 (bản sao run_probe_r14; fp_r14 của r14, ghi evidence/itacc_r14_fix_2/probes) — chạy 1 probe trên 1 cây, ghi bản ghi: sha256 probe, fingerprint sản phẩm (full + code, fp_r12fix),
lệnh, exit code, stdout/stderr thô, kết quả JSON của probe. Không ghi đè (tệp đích đã có → dừng).
  python -I run_probe.py.txt <probe a_ocr|...> <phia truoc|sau> <TREE> <NEW_FILES_LIST> [arg probe ...]"""
import hashlib
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8")
H = Path(__file__).resolve().parent
H13 = H
EV = H.parent
BASH = r"C:\Program Files\Git\usr\bin\bash.exe"
probe, phia, tree, newl = sys.argv[1:5]
extra = sys.argv[5:]
pf = H / f"probe_{probe}.py.txt"
psha = hashlib.sha256(pf.read_bytes()).hexdigest()
out = EV / "probes" / f"{probe}_{phia}"
out.mkdir(parents=True, exist_ok=False)


def fp(mode):
    r = subprocess.run([BASH, str(H.parents[1] / "itacc_r14_code_gaps" / "harness" / "fp_r14.sh.txt"), Path(tree).as_posix(), mode, Path(newl).as_posix()],
                       capture_output=True, text=True)
    return r.stdout.strip()


fpf, fpc = fp("full"), fp("code")
cmd = [sys.executable, "-I", str(pf), tree, *extra, str(out / "ket_qua.json")]
t0 = datetime.now(timezone.utc).isoformat(timespec="seconds")
r = subprocess.run(cmd, capture_output=True)
(out / "stdout.txt").write_bytes(r.stdout)
(out / "stderr.txt").write_bytes(r.stderr)
res = json.loads((out / "ket_qua.json").read_text(encoding="utf-8")) if (out / "ket_qua.json").exists() else None
rec = {"probe": probe, "phia": phia, "probe_file": pf.relative_to(EV.parents[1]).as_posix(), "probe_sha256": psha,
       "tree": tree, "fp_full": fpf, "fp_code": fpc, "lenh": " ".join(cmd), "bat_dau_utc": t0, "exit": r.returncode,
       "tong": res and res["tong"],
       "fail": [c["ca"] for c in (res or {"ca": []})["ca"] if c["ket_qua"] == "FAIL"]}
(out / "ban_ghi.json").write_text(json.dumps(rec, ensure_ascii=False, indent=1), encoding="utf-8")
print(json.dumps(rec, ensure_ascii=False, indent=1))
