def dsn(db, role=None):
    d = {**ADM, "dbname": db}
    if role:
        d.update(user=role, password=PW[role])
    return conninfo.make_conninfo(**d)


def clean_env(**kw):
    env = {k: v for k, v in os.environ.items() if not (k.startswith("PG") or k.endswith("DATABASE_URL")
                                                       or k in ("AUTH_MODE", "APPROVAL_ENFORCE", "DEPLOY"))}
    env.update(PYTHONIOENCODING="utf-8", **kw)
    return env


def apply(tree, db):
    r = subprocess.run([sys.executable, "scripts/apply_schema.py", "--roles"], cwd=tree, capture_output=True,
                       env=clean_env(DATABASE_URL=dsn(db)), timeout=600)
    err = r.stderr.decode("utf-8", "replace")
    return r.returncode, (r.stdout.decode("utf-8", "replace").strip().splitlines() or [""])[-1][:120], err[-300:]


def race(tree, bid, force, *callers):
    env = clean_env(**{f"RACE_DSN_{i + 1}": c[0] for i, c in enumerate(callers)})
    r = subprocess.run([sys.executable, str(HX / "race_upgrade.py.txt"), tree, str(bid), CAL, force,
                        *[c[1] for c in callers]], capture_output=True, env=env, timeout=300)
    line = [x for x in r.stdout.decode("utf-8", "replace").splitlines() if x.startswith("RACE_JSON ")]
    return json.loads(line[-1][10:]) if line else {"harness_err": r.stderr.decode("utf-8", "replace")[-400:]}

def case(ma, fn):
    try:
        ok, actual = fn()

        ok, actual = False, f"{type(e).__name__}: {str(e)[:240]}"

from pipelines import anchor as an  # noqa: E402  (chỉ dùng hàm định dạng — giống nhau ở 2 cây)

N = [0]


def seed(c, tag):
    N[0] += 1
    leaves = [{"k": f"q2:{SIDE}:{tag}:{N[0]}", "d": {"n": N[0]}, "h": "0" * 64}]
    as_of = datetime(2026, 10, 25, tzinfo=timezone.utc) + timedelta(seconds=N[0])
    root = hashlib.sha256(f"q2{SIDE}{tag}{N[0]}".encode()).hexdigest()
    sha = hashlib.sha256(an.manifest(int((as_of - an.EPOCH).total_seconds() * 1e6), leaves, root)).hexdigest()
    bid = c.execute("insert into anchor_batches (as_of, leaves, merkle_root, manifest_sha256, created_by) values"
                    " (%s, %s::jsonb, %s, %s, 'agent:anchor') returning id",
                    (as_of, json.dumps(leaves), root, sha)).fetchone()[0]
    resp = b"\xf0" + an._put_varuint(16) + b"\x22" * 16 + b"\x08" + b"\x00" + an.ATT_PENDING + \
        an._put_varuint(len(an._put_varuint(len(CAL)) + CAL.encode())) + an._put_varuint(len(CAL)) + CAL.encode()
    c.execute("insert into anchor_receipts (batch_id, provider, status, calendar, receipt, created_by) values"
              " (%s, 'opentimestamps', 'da_gui', %s, %s, 'agent:anchor')", (bid, CAL, an.ots_file(bytes.fromhex(sha), resp)))
    return bid, sha


def n_up(c, bid):
    return c.execute("select count(*) from anchor_receipts where batch_id = %s and status = 'da_nang_cap'",
                     (bid,)).fetchone()[0]


def snap(c):
    tabs = [r[0] for r in c.execute("select table_name from information_schema.tables where table_schema = 'public'"
                                    " and table_type = 'BASE TABLE' order by 1")]
    return ({t: c.execute(f'select count(*) from public."{t}"').fetchone()[0] for t in tabs},
            c.execute("select id, status, encode(sha256(coalesce(receipt, ''::bytea)), 'hex'), created_by"
                      " from anchor_receipts order by id").fetchall(),
            c.execute("select id, row_hash from agent_action_log order by id desc limit 1").fetchone(),
            set(c.execute("select grantee, table_name, privilege_type from information_schema.role_table_grants"
                          " where table_schema = 'public' and grantee <> 'postgres'").fetchall()))

def c_keep():

    diff = {k: (S0[0][k], S1[0].get(k)) for k in S0[0] if S1[0].get(k) != S0[0][k]}

def pair(st, c1, c2):
    def f():
        assert rc_n == 0, f"dựng DB mới lỗi: {tail_n} {err_n}"
        stage(adm_n, st)
        bid, _ = seed(adm_n, "D")
        res = race(ROOT, bid, "none", c1, c2)
        oks = [x.get("ok") for x in res.get("results", [])]
        news = sorted(bool(o and o.get("new")) for o in oks)
        ok = (n_up(adm_n, bid) == 1 and len(oks) == 2 and all(oks) and news == [False, True] and res.get("net") == 0)
        return ok, {"da_nang_cap": n_up(adm_n, bid), "race": res}
    return f


def blocked(st, caller):
    def f():
        assert rc_n == 0, f"dựng DB mới lỗi: {tail_n} {err_n}"
        stage(adm_n, st)
        bid, _ = seed(adm_n, "E")
        res = race(ROOT, bid, "none", caller)
        x = (res.get("results") or [{}])[0]
        rows_b = adm_n.execute("select count(*) from anchor_receipts where batch_id = %s", (bid,)).fetchone()[0]
        return "err" in x and x.get("gets") == 0 and rows_b == 1, {"race": res, "rows": rows_b}
    return f
