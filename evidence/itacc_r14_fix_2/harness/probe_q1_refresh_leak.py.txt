class Srv:
    def __init__(self, route):
        self.route, self.req = route, []
        srv = self

        class H(BaseHTTPRequestHandler):
            def _any(self):
                u = urllib.parse.urlsplit(self.path)
                n = int(self.headers.get("Content-Length") or 0)
                if n:
                    self.rfile.read(n)
                q = {k: v[0] for k, v in urllib.parse.parse_qs(u.query).items()}
                srv.req.append({"m": self.command, "p": u.path, "q": q})
                st, payload, hd = srv.route(self.command, u.path, q)
                data = payload if isinstance(payload, bytes) else json.dumps(payload).encode()
                self.send_response(st)
                for k, v in (hd or {}).items():
                    self.send_header(k, v)
                self.send_header("Content-Length", str(len(data)))
                self.end_headers()
                self.wfile.write(data)
            do_GET = do_POST = _any

            def log_message(self, *a):
                pass
        self.h = ThreadingHTTPServer(("127.0.0.1", 0), H)
        self.url = f"http://127.0.0.1:{self.h.server_address[1]}"
        threading.Thread(target=self.h.serve_forever, daemon=True).start()

    def close(self):
        self.h.shutdown()
        self.h.server_close()

def run_main(argv, env, mode="van_hanh"):
    old = dict(os.environ)
    os.environ.update(env)
    rollout._FORCE_MODE = mode
    o, e = io.StringIO(), io.StringIO()
    try:
        with contextlib.redirect_stdout(o), contextlib.redirect_stderr(e):
            try:
                rc = ma.main(argv)
            except SystemExit as x:
                rc = x.code
    finally:
        os.environ.clear()
        os.environ.update(old)
    return rc, o.getvalue() + e.getvalue()


def leaked(*texts):
    blob = "\n".join(texts)
    return [s[:16] + "…" for s in SECRETS if s in blob]


def direct_exc(ch, srv):
    """Gọi refresh_token() trực tiếp — chuỗi exception caller có thể in."""
    creds = ({"SHOPEE_PARTNER_ID": "100001", "SHOPEE_PARTNER_KEY": PKEY, "SHOPEE_SHOP_ID": "200002"}
             if ch == "shopee" else {"TIKTOK_APP_KEY": "fakeappkey", "TIKTOK_APP_SECRET": TSEC})
    try:
        ma.refresh_token(ch, creds, R_OLD, srv.url)
        return ""
    except Exception as x:  # noqa: BLE001
        return f"{x!s} | {x!r}"

def case(ma_, fn):
    try:
        ok, actual = fn()
    except Exception as e:  # noqa: BLE001

    rows.append({"ca": ma_, "ket_qua": "PASS" if ok else "FAIL", "actual": actual})


def refresh_err(ch, payload, status=200, headers=None, want=None):
    def f():
        auth = ma.SP_AUTH_REFRESH if ch == "shopee" else ma.TT_AUTH_REFRESH
        s = Srv(lambda m, p, q: (status, payload, headers) if p == auth else (404, {}, None))
        kho = store(ch, f"{ch}{len(rows)}")
        before = kho.read_bytes()
        try:
            rc, text = run_main(["--channel", ch, "--refresh"], env_for(ch, s, kho))
            n_main = len(s.req)
            exc = direct_exc(ch, s)
        finally:
            s.close()
        lk = leaked(text, exc)
        ok = (rc == 1 and n_main == 1 and kho.read_bytes() == before and not lk
              and (want is None or want in text))
        return ok, {"rc": rc, "requests": n_main, "lo": lk, "stdout": text[-220:].replace(R_OLD, "<R_OLD>")
                    .replace(A_OLD, "<A_OLD>").replace(PKEY, "<PKEY>").replace(TSEC, "<TSEC>"),
                    "exc": exc[:160].replace(R_OLD, "<R_OLD>").replace(A_OLD, "<A_OLD>").replace(PKEY, "<PKEY>")
                    .replace(TSEC, "<TSEC>")}
    return f
