"""Tổng hợp HIỆN HÀNH r11 cho 96 mục: 90 bản ghi r10 giữ nguyên + 6 bản ghi r11 thay đúng 6 id.
Ghi evidence/itacc_r11/source_map.json + tong_hop.json (r10/tong_hop.json giữ làm lịch sử).
Kiểm: đủ đúng 96 id; đúng 1 nguồn/id; tổng 3 nhóm = 96; ĐẠT không có FAIL/NOT_RUN; FAIL → BI_CHAN; BI_CHAN có
nguyên nhân + việc gỡ; CHUA_KIEM có NOT_RUN kèm lý do; evidence tồn tại + hash khớp (alias .py→.py.txt cho harness
r10 đã đổi đuôi); bản ghi r11: hash tiêu chí = khóa, supersedes khớp hash bản ghi r10, mọi file log ca tồn tại và
nằm dưới evidence/itacc_r11 (không dùng log tái tạo r10). python -I aggregate_r11.py.txt → exit 1 nếu vi phạm."""
import collections
import hashlib
import json
import sys
from pathlib import Path

sys.stdout.reconfigure(encoding="utf-8")
ROOT = Path(__file__).resolve().parents[3]
R10, R11 = ROOT / "evidence" / "itacc_r10", ROOT / "evidence" / "itacc_r11"
khung = {i["id"]: i for i in json.loads((R10 / "criteria" / "khung.json").read_text(encoding="utf-8"))["items"]}
NEW = ("marketing-03.2", "marketing-05.1", "marketing-05.2", "marketing-06.1", "marketing-06.2", "legal_hr-07")


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def resolve(path_s):
    p = Path(path_s)
    p = p if p.is_absolute() else ROOT / p
    if not p.exists() and p.suffix == ".py" and p.with_name(p.name + ".txt").exists():
        p = p.with_name(p.name + ".txt")
    return p


errs, rows, smap, nref = [], [], {}, 0
for fid, k in khung.items():
    src = (R11 if fid in NEW else R10) / "items" / f"{fid}.json"
    smap[fid] = {"luot": "r11" if fid in NEW else "r10", "path": str(src.relative_to(ROOT)).replace("\\", "/"), "sha256": sha(src)}
    r = json.loads(src.read_text(encoding="utf-8"))
    st, ca = r.get("trang_thai"), r.get("ca") or []
    kq = collections.Counter(c.get("ket_qua") for c in ca)
    if r.get("id") != fid:
        errs.append(f"{fid}: id trong file = {r.get('id')}")
    if st not in ("DAT", "CHUA_KIEM", "BI_CHAN"):
        errs.append(f"{fid}: trạng thái lạ {st}")
    if not ca:
        errs.append(f"{fid}: không có ca")
    if st == "DAT" and (kq.get("FAIL") or kq.get("NOT_RUN") or kq.get("LOI_HARNESS") or not kq.get("PASS")):
        errs.append(f"{fid}: DAT nhưng ca {dict(kq)}")
    if kq.get("FAIL") and st != "BI_CHAN":
        errs.append(f"{fid}: có FAIL nhưng {st}")
    if st == "BI_CHAN" and not ((r.get("nguyen_nhan") or "").strip() and (r.get("can_de_go") or "").strip()):
        errs.append(f"{fid}: BI_CHAN thiếu nguyên nhân/việc gỡ")
    if st == "CHUA_KIEM" and not any(c.get("ket_qua") in ("NOT_RUN", "LOI_HARNESS") and (c.get("ly_do_not_run") or c.get("ma")) for c in ca):
        errs.append(f"{fid}: CHUA_KIEM không có ca NOT_RUN")
    for e in r.get("evidence") or []:
        nref += 1
        p = resolve(e.get("path", ""))
        if not p.exists():
            errs.append(f"{fid}: evidence không tồn tại {e.get('path')}")
        elif e.get("sha256") and sha(p) != e["sha256"]:
            errs.append(f"{fid}: hash evidence lệch {e['path']}")
    if fid in NEW:
        cp = ROOT / "evidence" / "itacc_r11" / r["tieu_chi_file"].split("itacc_r11/", 1)[1]
        lock = cp.with_name(cp.name.replace(".json", ".lock.txt")).read_text(encoding="utf-8").split()[1]
        if not (sha(cp) == r["tieu_chi_sha256"] == lock):
            errs.append(f"{fid}: hash tiêu chí ≠ khóa")
        old = ROOT / r["supersedes"]["path"]
        if sha(old) != r["supersedes"]["sha256"]:
            errs.append(f"{fid}: supersedes hash lệch (bản ghi r10 bị sửa?)")
        for c in ca:
            for b in c["actual"]["buoc"]:
                for kk in ("stdout", "stderr"):
                    if not b[kk].startswith("evidence/itacc_r11/runs/") or not (ROOT / b[kk]).exists():
                        errs.append(f"{fid}: {c['ma']} log {kk} không phải log gốc r11 hiện có: {b[kk]}")
    rows.append({"id": fid, "khoi": k["khoi"], "giai_doan": k["giai_doan"], "pham_vi": k["pham_vi"], "ten": k["ten"],
                 "luot": smap[fid]["luot"], "trang_thai": st, "muc_kiem": r.get("muc_kiem"), "ca": dict(kq),
                 "nguyen_nhan": (r.get("nguyen_nhan") or "").split(":")[0].split(" ")[0].strip() if st == "BI_CHAN" else "",
                 "nguyen_nhan_day_du": r.get("nguyen_nhan") or "", "loi": r.get("loi_san_pham") or []})
extra = sorted(p.stem for p in (R11 / "items").glob("*.json") if p.stem not in NEW)
if extra:
    errs.append(f"bản ghi r11 ngoài 6 mục: {extra}")
tot = collections.Counter(r["trang_thai"] for r in rows)
by_khoi, by_scope = collections.defaultdict(collections.Counter), collections.defaultdict(collections.Counter)
for r in rows:
    by_khoi[r["khoi"]][r["trang_thai"]] += 1
    by_scope[f"AI {r['giai_doan']}" if r["pham_vi"] == "tinh_nang_ai" else "7 hạng mục khác"][r["trang_thai"]] += 1
cause = collections.Counter(r["nguyen_nhan"] for r in rows if r["trang_thai"] == "BI_CHAN")
r10_tot = json.loads((R10 / "tong_hop.json").read_text(encoding="utf-8"))["tom_tat"]["tong"]
summary = {"mau_so": len(khung), "nguon_r10": sum(v["luot"] == "r10" for v in smap.values()),
           "nguon_r11": sum(v["luot"] == "r11" for v in smap.values()), "tong": dict(tot),
           "theo_khoi": {k: dict(v) for k, v in by_khoi.items()}, "theo_pham_vi": {k: dict(v) for k, v in by_scope.items()},
           "nguyen_nhan_bi_chan": dict(cause), "checkpoint_r10_lich_su": r10_tot, "so_tham_chieu_evidence": nref,
           "loi_nhat_quan": errs}
if len(rows) != 96 or sum(tot.values()) != 96:
    errs.append("không đủ 96")
(R11 / "source_map.json").write_text(json.dumps({"_doc": "Nguồn kết quả hiện hành từng id (r11 thay đúng 6 id; 90 id kế thừa r10 không sửa).", "items": smap}, ensure_ascii=False, indent=1), encoding="utf-8")
(R11 / "tong_hop.json").write_text(json.dumps({"tom_tat": summary, "items": rows}, ensure_ascii=False, indent=1), encoding="utf-8")
print(json.dumps(summary, ensure_ascii=False, indent=1))
sys.exit(1 if errs else 0)
