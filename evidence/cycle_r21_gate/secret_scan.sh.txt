#!/bin/bash
# r21 secret scan: evidence/cycle_r21_gate + dòng thêm trong diff từ Base 43263c5 + mật khẩu container samsam-ci-repro
# (sinh ở scratchpad) trong tệp theo dõi + tệp chưa theo dõi của r21 (evidence/cycle_r21_gate). Ghi secret_scan.txt.
#   bash secret_scan.sh.txt <SP>
SP="$1"; E=/d/code/samsam/evidence/cycle_r21_gate; OUTF=$E/secret_scan.txt
cd /d/code/samsam || exit 2
PW=$(sed -nE "s/^export CT_PW='([^']+)'.*/\1/p" "$SP/secret/admin.sh")
[ ${#PW} -ge 20 ] || { echo "không đọc được mật khẩu container"; exit 2; }
{ echo "## secret scan $E + diff từ Base 43263c5 (dòng thêm) — $(date -Iseconds)"
  echo "### grep -rEn 'postgres(ql)?://[^ ]*:[^*@]+@' → (rỗng = đạt)"
  grep -rEn 'postgres(ql)?://[^ ]*:[^*@]+@' "$E" --exclude=secret_scan.sh.txt; git diff 43263c5 -- . | grep '^+' | grep -En 'postgres(ql)?://[^ ]*:[^*@]+@'
  echo "### grep -rn 'sk-[o]r' (tiền tố khóa OpenRouter) → (rỗng = đạt)"
  grep -rn 'sk-[o]r' "$E" --exclude=secret_scan.sh.txt; git diff 43263c5 -- . | grep '^+' | grep -n 'sk-[o]r'
  echo "### mật khẩu container trong tệp theo dõi + evidence/cycle_r21_gate → (rỗng = đạt)"
  { git ls-files; git ls-files --others --exclude-standard -- evidence/cycle_r21_gate; } | tr '\n' '\0' | xargs -0 grep -lF -- "$PW" 2>/dev/null
  echo "### đối chứng dương (phải có ≥1): tệp scratchpad chứa mật khẩu"
  grep -lF -- "$PW" "$SP/secret/admin.sh" | wc -l
  echo "EXIT 0"; } > "$SP/secret_scan.tmp"
mv "$SP/secret_scan.tmp" "$OUTF"
cat "$OUTF"
