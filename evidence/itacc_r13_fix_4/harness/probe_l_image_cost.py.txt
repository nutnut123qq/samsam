"""PROBE r13-fix4 L v1 — F2: adapter ảnh nền `pipelines.image_provider_openrouter.background` phải GIỮ chi phí provider đã
báo khi ảnh trả về hỏng, ghi đúng MỘT dòng sổ chi phí với trạng thái thất bại, và cổng tháng (`llm_gateway._month_spend`)
thực sự cộng khoản đó. HTTP thay bằng httpx.MockTransport (0 mạng); DB thay bằng conn giả trong bộ nhớ đi qua ĐÚNG
`gw._record` + `gw._month_spend` thật (câu SQL thật, tham số thật). Cùng mã/input/assertion snapshot r13 chưa vá + bản vá.
  python -I probe_l_image_cost.py.txt <ROOT> <out.json>
Tiêu chí (khóa trước khi vá):
- ảnh hợp lệ + usage.cost 0.40 → trả ảnh; sổ 1 dòng cost 0.40 ok=True; chi tiêu tháng agent `image` và tính năng
  marketing-02.1 = 0.40.
- HTTP 200 + usage.cost 0.40 nhưng b64 hỏng / data rỗng / thiếu data / byte không phải ảnh → ImageProviderError; sổ ĐÚNG 1
  dòng cost 0.40 ok=False; chi tiêu tháng (agent + tính năng) = 0.40.
- HTTP 200 không đọc được usage (b64 hỏng không có usage / thân không phải JSON) → 1 dòng cost = mức dự phòng hiện có
  (`image_cost_fallback_usd` cấu hình hoặc 0.25) > 0, ok=False.
- đối chứng: 502 hai lần (không tính phí theo tài liệu) → 1 dòng cost 0 ok=False; nút tắt tính năng → 0 request, 0 dòng."""
import base64
import io
import json
import os
import sys
import time

ROOT, OUT = sys.argv[1], sys.argv[2]
sys.stdout.reconfigure(encoding="utf-8")
sys.path.insert(0, ROOT)
os.chdir(ROOT)
for k in list(os.environ):
    if k.endswith("DATABASE_URL") or k.startswith(("PG", "LLM_")) or k in ("OPENROUTER_API_KEY", "IMAGE_MODEL"):
        os.environ.pop(k, None)
import httpx  # noqa: E402
import psycopg  # noqa: E402
from PIL import Image  # noqa: E402

psycopg.connect = lambda *a, **k: (_ for _ in ()).throw(RuntimeError("probe: cấm DB"))
from pipelines import image_provider_openrouter as orp  # noqa: E402
from pipelines import llm_gateway as gw  # noqa: E402
from pipelines import rollout  # noqa: E402
from pipelines.image_studio import ImageProviderError  # noqa: E402

rollout._FORCE_MODE = "van_hanh"          # giai đoạn ép như conftest — không đọc sổ DB
time.sleep = lambda s: None
LEDGER = []


class Conn:
    def execute(self, sql, params=None):
        self.res = None
        if sql.startswith("insert into llm_usage"):
            LEDGER.append(dict(zip(("agent_id", "model", "tin", "tout", "cost_usd", "ok", "error", "feature_id"),
                                   params)))
        elif "sum(cost_usd)" in sql:
            col = "feature_id" if "where feature_id" in sql else "agent_id"
            self.res = (sum(float(r["cost_usd"]) for r in LEDGER if r[col] == params[0]),)
        return self

    def fetchone(self):
        return self.res

    def commit(self):
        pass

    def close(self):
        pass


gw._connect = lambda: Conn()
os.environ.update({"OPENROUTER_API_KEY": "probe-khoa-gia", "IMAGE_MODEL": "probe/model-gia",
                   "LLM_PROVIDER_ONLY": "prov-probe"})
FALLBACK = float(gw._config().get("image_cost_fallback_usd") or 0.25)
_b = io.BytesIO()
Image.new("RGB", (32, 32), (10, 90, 40)).save(_b, "PNG")
PNG_B64 = base64.b64encode(_b.getvalue()).decode()
rows = []


def run(responses):
    """responses: list httpx.Response theo thứ tự → (kết quả, số request)."""
    reqs = []

    def handler(req):
        reqs.append(req.url.path)
        return responses[min(len(reqs), len(responses)) - 1]
    orp._image_http = lambda: httpx.Client(transport=httpx.MockTransport(handler))
    LEDGER.clear()
    try:
        img, _meta = orp.background("nền vườn sâm (probe)", (32, 32))
        return f"ảnh {img.size}", len(reqs)
    except ImageProviderError as e:
        return f"ImageProviderError: {e}", len(reqs)


def spends():
    return gw._month_spend("image"), gw._month_spend("marketing-02.1", gw._FEATURE_SPEND_SQL)


def case(ma, fn):
    try:
        ok, actual = fn()
    except Exception as e:  # noqa: BLE001
        ok, actual = False, f"{type(e).__name__}: {str(e)[:200]}"
    rows.append({"ca": ma, "ket_qua": "PASS" if ok else "FAIL", "actual": actual, "so_chi_phi": list(LEDGER)})


def billed_fail(body, want_cost, raw=False):
    def f():
        resp = httpx.Response(200, content=body) if raw else httpx.Response(200, json=body)
        out, n = run([resp])
        sa, sf = spends()
        ok = (out.startswith("ImageProviderError") and n == 1 and len(LEDGER) == 1
              and abs(float(LEDGER[0]["cost_usd"]) - want_cost) < 1e-9 and LEDGER[0]["ok"] is False
              and abs(sa - want_cost) < 1e-9 and abs(sf - want_cost) < 1e-9 and want_cost > 0)
        return ok, {"out": out, "spend_agent": sa, "spend_feature": sf}
    return f


def c_valid():
    out, n = run([httpx.Response(200, json={"data": [{"b64_json": PNG_B64}], "usage": {"cost": 0.40}})])
    sa, sf = spends()
    ok = (out == "ảnh (32, 32)" and len(LEDGER) == 1 and abs(float(LEDGER[0]["cost_usd"]) - 0.40) < 1e-9
          and LEDGER[0]["ok"] is True and abs(sa - 0.40) < 1e-9 and abs(sf - 0.40) < 1e-9)
    return ok, {"out": out, "spend_agent": sa, "spend_feature": sf}


def c_502():
    out, n = run([httpx.Response(502, json={}), httpx.Response(502, json={})])
    ok = out.startswith("ImageProviderError") and n == 2 and len(LEDGER) == 1 and float(LEDGER[0]["cost_usd"]) == 0.0 \
        and LEDGER[0]["ok"] is False
    return ok, {"out": out, "requests": n}


def c_kill():
    os.environ["LLM_KILL_FEATURES"] = "marketing-02.1"
    try:
        out, n = run([httpx.Response(200, json={"data": [{"b64_json": PNG_B64}], "usage": {"cost": 0.40}})])
    finally:
        os.environ.pop("LLM_KILL_FEATURES", None)
    return out.startswith("ImageProviderError") and n == 0 and LEDGER == [], {"out": out, "requests": n}


case("anh_hop_le_cost_040", c_valid)
case("b64_hong_giu_cost_040", billed_fail({"data": [{"b64_json": "!bad-base64!"}], "usage": {"cost": 0.40}}, 0.40))
case("data_rong_giu_cost_040", billed_fail({"data": [], "usage": {"cost": 0.40}}, 0.40))
case("thieu_data_giu_cost_040", billed_fail({"usage": {"cost": 0.40}}, 0.40))
case("byte_khong_phai_anh_giu_cost_040_ok_false",
     billed_fail({"data": [{"b64_json": base64.b64encode(b"khong-phai-anh").decode()}], "usage": {"cost": 0.40}}, 0.40))
case("b64_hong_khong_usage_du_phong", billed_fail({"data": [{"b64_json": "!bad-base64!"}]}, FALLBACK))
case("than_khong_phai_json_du_phong", billed_fail(b"<html>not json</html>", FALLBACK, raw=True))
case("doi_chung_502_khong_tinh_phi", c_502)
case("doi_chung_nut_tat_0_request", c_kill)
out = {"probe": "l_image_cost_v1", "root": ROOT, "fallback_usd": FALLBACK, "ca": rows,
       "tong": {k: sum(r["ket_qua"] == k for r in rows) for k in ("PASS", "FAIL")}}
open(OUT, "x", encoding="utf-8").write(json.dumps(out, ensure_ascii=False, indent=1, default=str))
for r in rows:
    print(r["ket_qua"], r["ca"], str(r["actual"])[:160])
print("TONG", out["tong"])
sys.exit(1 if out["tong"]["FAIL"] else 0)
